00개요 · 실제 저장소와 직렬 통신

제13장의 에코 제어기는 조합 전달 회로였습니다. 제14장은 실제 UART(유아트, 비동기 직렬 통신)의 RX/TX FIFO(피포, 선입선출 버퍼)와 직렬 송수신 회로를 검증합니다. 수락, 포화, 동시 읽기·쓰기, 초과 수신 및 리셋을 구분합니다.

reference_rtl/uart_core.v · Verilog
// (변경) 8N1 PHY 내부에 깊이 16 RX/TX FIFO를 넣어 byte stream ready/valid를 보존.
// 1바이트 UART 수신 레지스터와 FIFO는 별개. Controller는 stop-and-wait 정책을 유지.
module uart_core #(
    parameter integer CLK_HZ=100000000,BAUD=115200
)(
    input wire clk,resetn,uart_rxd_i,
    output wire uart_txd_o,
    output wire [7:0] rx_out_data,
    output wire rx_out_valid,
    input wire rx_out_ready,
    input wire [7:0] tx_in_data,
    input wire tx_in_valid,
    output wire tx_in_ready,
    output wire rx_framing_error_o,rx_overrun_o
);
    wire [7:0] phy_rx_data;
    wire phy_rx_valid,phy_rx_ready;
    wire [7:0] phy_tx_data;
    wire phy_tx_valid,phy_tx_ready;
    uart_rx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) rx(
        .clk(clk),.resetn(resetn),.rx_i(uart_rxd_i),
        .out_data_o(phy_rx_data),.out_valid_o(phy_rx_valid),
        .out_ready_i(phy_rx_ready),.framing_error_o(rx_framing_error_o),
        .overrun_o(rx_overrun_o));
    uart_tx #(.CLK_HZ(CLK_HZ),.BAUD(BAUD)) tx(
        .clk(clk),.resetn(resetn),.in_data_i(phy_tx_data),
        .in_valid_i(phy_tx_valid),.in_ready_o(phy_tx_ready),.tx_o(uart_txd_o));

    reg [7:0] rx_fifo[0:15];
    reg [3:0] rx_wptr,rx_rptr;
    reg [4:0] rx_count;
    wire rx_pop=rx_out_valid && rx_out_ready;
    wire rx_push=phy_rx_valid && phy_rx_ready;
    assign rx_out_valid=(rx_count!=0);
    assign rx_out_data=rx_fifo[rx_rptr];
    assign phy_rx_ready=resetn && ((rx_count<16) || rx_pop);

    reg [7:0] tx_fifo[0:15];
    reg [3:0] tx_wptr,tx_rptr;
    reg [4:0] tx_count;
    wire tx_pop=phy_tx_ready && phy_tx_valid;
    wire tx_push=tx_in_valid && tx_in_ready;
    assign tx_in_ready=resetn && ((tx_count<16) || tx_pop);
    assign phy_tx_valid=(tx_count!=0);
    assign phy_tx_data=tx_fifo[tx_rptr];

    always @(posedge clk)begin
        if(!resetn)begin
            rx_wptr<=0;rx_rptr<=0;rx_count<=0;
            tx_wptr<=0;tx_rptr<=0;tx_count<=0;
        end else begin
            if(rx_push)begin
                rx_fifo[rx_wptr]<=phy_rx_data;
                rx_wptr<=rx_wptr+1'b1;
            end
            if(rx_pop)rx_rptr<=rx_rptr+1'b1;
            case({rx_push,rx_pop})
                2'b10:rx_count<=rx_count+1'b1;
                2'b01:rx_count<=rx_count-1'b1;
                default:begin end
            endcase
            if(tx_push)begin
                tx_fifo[tx_wptr]<=tx_in_data;
                tx_wptr<=tx_wptr+1'b1;
            end
            if(tx_pop)tx_rptr<=tx_rptr+1'b1;
            case({tx_push,tx_pop})
                2'b10:tx_count<=tx_count+1'b1;
                2'b01:tx_count<=tx_count-1'b1;
                default:begin end
            endcase
        end
    end
endmodule
reference_rtl/uart_rx.v · Verilog
// (추가) 8N1 수신. RX 2FF, start midpoint/bit center, hold/overrun.
module uart_rx #(
    parameter integer CLK_HZ=100000000, BAUD=115200
)(
    input wire clk,resetn,rx_i,
    output reg [7:0] out_data_o,
    output reg out_valid_o,
    input wire out_ready_i,
    output reg framing_error_o,overrun_o
);
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD;
    localparam integer HALF=CPB/2;
    localparam [1:0] IDLE=0,START=1,DATA=2,STOP=3;
    reg [1:0] state;
    reg rx_meta,rx_sync;
    reg [31:0] count_q;
    reg [2:0] bit_q;
    reg [7:0] shift_q;
    always @(posedge clk)begin
        if(!resetn)begin
            state<=IDLE;rx_meta<=1;rx_sync<=1;count_q<=0;bit_q<=0;shift_q<=0;
            out_data_o<=0;out_valid_o<=0;framing_error_o<=0;overrun_o<=0;
        end else begin
            rx_meta<=rx_i;rx_sync<=rx_meta;
            if(out_valid_o && out_ready_i)out_valid_o<=0;
            case(state)
                IDLE:if(!rx_sync)begin count_q<=HALF-1;state<=START;end
                START:if(count_q==0)begin
                    if(!rx_sync)begin count_q<=CPB-1;bit_q<=0;state<=DATA;end
                    else state<=IDLE;
                end else count_q<=count_q-1'b1;
                DATA:if(count_q==0)begin
                    shift_q[bit_q]<=rx_sync;count_q<=CPB-1;
                    if(bit_q==7)state<=STOP;
                    else bit_q<=bit_q+1'b1;
                end else count_q<=count_q-1'b1;
                STOP:if(count_q==0)begin
                    state<=IDLE;
                    if(!rx_sync)framing_error_o<=1;
                    else if(!out_valid_o || out_ready_i)begin
                        out_data_o<=shift_q;out_valid_o<=1;
                    end else overrun_o<=1;
                end else count_q<=count_q-1'b1;
                default:state<=IDLE;
            endcase
        end
    end
    // synthesis translate_off
    initial if(CPB<4)$fatal(1,"Unsupported UART clock/baud ratio");
    // synthesis translate_on
endmodule
reference_rtl/uart_tx.v · Verilog
// (추가) 8N1 송신. valid/ready handshake, 10bit start/data/stop.
module uart_tx #(
    parameter integer CLK_HZ=100000000, BAUD=115200
)(
    input wire clk,resetn,
    input wire [7:0] in_data_i,
    input wire in_valid_i,
    output wire in_ready_o,
    output reg tx_o
);
    localparam integer CPB=(CLK_HZ+BAUD/2)/BAUD;
    reg sending;
    reg [31:0] tick_q;
    reg [3:0] bit_q;
    reg [9:0] frame_q;
    assign in_ready_o=resetn && !sending;
    always @(posedge clk)begin
        if(!resetn)begin
            sending<=0;tick_q<=0;bit_q<=0;frame_q<=10'h3ff;tx_o<=1;
        end else if(!sending)begin
            tx_o<=1;
            if(in_valid_i)begin
                sending<=1;tick_q<=CPB-1;bit_q<=0;
                frame_q<={1'b1,in_data_i,1'b0};tx_o<=0;
            end
        end else if(tick_q!=0)tick_q<=tick_q-1'b1;
        else begin
            tick_q<=CPB-1;frame_q<={1'b1,frame_q[9:1]};
            if(bit_q==9)begin sending<=0;tx_o<=1;end
            else begin bit_q<=bit_q+1'b1;tx_o<=frame_q[1];end
        end
    end
    // synthesis translate_off
    initial if(CPB<4)$fatal(1,"Unsupported UART clock/baud ratio");
    // synthesis translate_on
endmodule

FIFO의 4비트 포인터는 0~15를 순환하고 5비트 count는 0~16을 나타냅니다. RX FIFO와 수신기의 out_data_o 보류 레지스터, TX FIFO와 송신기의 frame_q는 서로 다른 저장 공간입니다. 메모리 배열을 리셋할 때 모두 0으로 지우지 않고 count와 포인터를 초기화합니다. 따라서 valid=0일 때 rx_out_data 값은 검사하지 않습니다.

02FIFO 계약 · 포화와 동시 입출력

실제 식의미
rx_out_valid = (rx_count != 0)RX 읽을 데이터 존재
rx_pop = rx_out_valid && rx_out_readyRX 소비
phy_rx_ready = resetn && ((rx_count < 16) || rx_pop)RX full이어도 pop과 함께 push 허용
tx_in_ready = resetn && ((tx_count < 16) || tx_pop)TX 공간 또는 동시 pop
tx_pop = phy_tx_ready && phy_tx_validTX PHY로 전달
pushpopcount 변화
00유지
10+1
01-1
11유지

가득 찬 FIFO에 push와 pop이 동시에 발생하면 이전 맨 앞 데이터를 전달하면서 같은 주소에 새 데이터를 기록할 수 있습니다. 비차단 대입에 의해 기존 읽기값으로 거래가 성립합니다. 단순히 18개를 보내는 것으로 포화 시험을 완료했다고 판정하지 않고 실제 stall과 full 동시 교체를 확인합니다.

03직렬 비트 시간과 독립 관찰

TB 클록은 10ns(100MHz), 파라미터 CLK_HZ=100000000, BAUD=10000000입니다. CPB(클록/비트)는 10, 한 비트는 100ns, 8N1 프레임은 10비트로 1000ns입니다. 실제 보드 기본 115200bps와 다릅니다. 송신기는 프레임 종료 후 다음 상승 에지에서 새 프레임을 시작할 수 있어 연속 시작 간격은 101클록입니다.

구간기대값TX 시작 하강 에지 이후 관찰
Start045ns
D0~D7LSB부터 8비트145~845ns, 100ns 간격
Stop1945ns

원본 uart_rx를 정답 모니터로 재사용하면 송수신기의 공통 해석 오류를 놓칠 수 있습니다. TB는 txd 하강 에지에서 시작하여 각 비트 중앙 부근에서 직접 샘플링합니다. 18개 프레임 뒤 200클록 동안 TX가 HIGH인지 확인합니다. 정상 8N1 경로의 표본 검사이며 모든 순간 글리치·보레이트 오차를 측정하는 시험은 아닙니다.

04검증 시나리오 · 실제 경계 발생 확인

시험입력기대 결과
T1 TX0x40~0x51, 18개18 수락/18 복조, stall 및 full 교체 발생
T2 RX0x80~0x8F, 16개full·대기 유지·16개 순서·empty
T3 RX 초과0xA0~0xB1, 18개FIFO 16+PHY 1 보존, 마지막 0xB1 유실, overrun=1
T4 리셋RX C3 보류, TX 00/11/22RX 1개 및 TX FIFO 2개와 송신 중 프레임 폐기

T3에서 첫 pop은 PHY에 보류된 17번째 바이트를 FIFO로 밀어 넣으므로 count 16이 유지됩니다. 이후 16개가 더 배출되면 empty가 됩니다. overrun은 리셋까지 유지되는 상태입니다. T4는 RX valid가 1이고 TX가 실제로 송신 중이며 FIFO에 2개가 남았음을 확인한 뒤 리셋합니다. TX 핀은 다음 상승 에지에서 idle HIGH가 됩니다. 리셋 이후 200클록 동안 잔류 데이터가 나타나지 않는지 확인합니다.

순서와 수락 판정은 공개 포트, full 경계 발생은 내부 count/push/pop으로 확인합니다. 내부 접근은 검증용 계층 참조이며 합성 설계 변경이 아닙니다. RX의 framing error 발생 자체는 별도 시험 범위입니다.

tb/ch14/tb_uart_fifo_flow.sv · SystemVerilog
`timescale 1ns/1ps
// 제14장: 실제 uart_core의 RX/TX 깊이 16 FIFO를 공개 포트로 검증.
// 내부 카운터는 교육용 관찰에만 사용하며, 정답 판정은 포트 계약 중심.
module tb_uart_fifo_flow;
    localparam integer CLK_HZ=100000000, BAUD=10000000, CPB=10;
    reg clk=0, resetn=0, rxd=1;
    wire txd;
    wire [7:0] rx_out_data;
    wire rx_out_valid, tx_in_ready, fe, ov;
    reg rx_out_ready=0, tx_in_valid=0;
    reg [7:0] tx_in_data=0;
    integer accepted=0, received=0, rx_checked=0;
    integer phase=0, tx_stalls=0, tx_full_swap=0, rx_full_swap=0;
    reg rx_held=0;
    reg [7:0] held_data;
    always #5 clk=~clk;

    uart_core #(.CLK_HZ(CLK_HZ), .BAUD(BAUD)) dut (
        .clk(clk), .resetn(resetn), .uart_rxd_i(rxd), .uart_txd_o(txd),
        .rx_out_data(rx_out_data), .rx_out_valid(rx_out_valid),
        .rx_out_ready(rx_out_ready), .tx_in_data(tx_in_data),
        .tx_in_valid(tx_in_valid), .tx_in_ready(tx_in_ready),
        .rx_framing_error_o(fe), .rx_overrun_o(ov)
    );
    // (변경) 실제 상승 에지의 거래를 독립 집계한다.
    // 내부 count는 경계 발생 및 용량 검사의 보조 증거로 사용한다.
    always @(posedge clk) begin
        if (!resetn) rx_held=0;
        else begin
            if ((^dut.rx_count) === 1'bx || (^dut.tx_count) === 1'bx ||
                dut.rx_count>16 || dut.tx_count>16)
                $fatal(1,"FAIL FIFO_COUNT");
            if (phase==1 && tx_in_valid) begin
                if (tx_in_ready === 1'b0) tx_stalls++;
                if (tx_in_ready === 1'b1) begin
                    if (accepted>=18 || tx_in_data !== (8'h40+8'(accepted)))
                        $fatal(1,"FAIL TX_ACCEPT_ORDER");
                    accepted++;
                    if (dut.tx_count==16 && dut.tx_pop) tx_full_swap++;
                end
            end
            if (rx_held && (rx_out_valid !== 1'b1 || rx_out_data !== held_data))
                $fatal(1,"FAIL RX_HOLD");
            rx_held=(rx_out_valid === 1'b1 && rx_out_ready === 1'b0);
            if (rx_held) held_data=rx_out_data;
            if ((phase==2 || phase==3) &&
                rx_out_valid === 1'b1 && rx_out_ready === 1'b1) begin
                if (rx_checked>=((phase==2)?16:17) ||
                    rx_out_data !== ((phase==2?8'h80:8'ha0)+8'(rx_checked)))
                    $fatal(1,"FAIL RX_ORDER phase=%0d idx=%0d",phase,rx_checked);
                rx_checked++;
                if (phase==3 && dut.rx_count==16 && dut.rx_push)
                    rx_full_swap++;
            end
        end
    end

    // 입력 바이트를 handshake 때만 수락한다.
    task automatic push_tx(input [7:0] data);
        integer guard;
        begin
            @(negedge clk);
            tx_in_data=data; tx_in_valid=1;
            guard=0;
            while (tx_in_ready !== 1'b1) begin
                @(negedge clk);
                guard=guard+1;
                if (guard>2000) $fatal(1,"FAIL TX_READY_TIMEOUT");
            end
            @(posedge clk);
            if (tx_in_ready !== 1'b1) $fatal(1,"FAIL TX_HANDSHAKE");
            @(negedge clk);
            tx_in_valid=0;
        end
    endtask

    // 8N1: start(0), D0~D7, stop(1). LSB부터 전송한다.
    task automatic send_rx(input [7:0] data);
        integer i;
        begin
            @(negedge clk);
            rxd=0;
            repeat(CPB) @(negedge clk);
            for(i=0;i<8;i=i+1) begin
                rxd=data[i];
                repeat(CPB) @(negedge clk);
            end
            rxd=1;
            repeat(CPB) @(negedge clk);
        end
    endtask

    // (변경) DUT의 uart_rx를 재사용하지 않는 독립 8N1 복조기.
    // TX 시작 에지 이후 각 비트 중앙 부근에서 직접 관찰한다.
    reg [7:0] decoded;
    integer frame_index, bit_index;
    initial begin
        wait(phase==1);
        for(frame_index=0;frame_index<18;frame_index++) begin
            @(negedge txd);
            repeat(CPB/2) @(negedge clk);
            if(txd !== 1'b0) $fatal(1,"FAIL TX_START");
            for(bit_index=0;bit_index<8;bit_index++) begin
                repeat(CPB) @(negedge clk);
                decoded[bit_index]=txd;
            end
            repeat(CPB) @(negedge clk);
            if(txd !== 1'b1) $fatal(1,"FAIL TX_STOP");
            if(received>=accepted || decoded !== (8'h40+8'(frame_index)))
                $fatal(1,"FAIL TX_ORDER idx=%0d got=%h",frame_index,decoded);
            received++;
        end
    end

    task automatic drain_rx(input integer total);
        integer k;
        begin
            for(k=0;k<total;k++) begin
                @(negedge clk);
                if(rx_out_valid !== 1'b1) $fatal(1,"FAIL RX_EMPTY_EARLY");
                rx_out_ready=1;
                @(negedge clk); rx_out_ready=0;
            end
            if(rx_out_valid !== 1'b0 || rx_checked!=total)
                $fatal(1,"FAIL RX_DRAIN");
        end
    endtask

    initial begin
        #1000000;
        $fatal(1,"FAIL GLOBAL_TIMEOUT");
    end

`ifdef VCD
    initial begin
        $dumpfile("ch14_uart_fifo.vcd");
        $dumpvars(0,tb_uart_fifo_flow);
    end
`endif
    integer i,guard;
    initial begin
        repeat(5) @(negedge clk);
        if(tx_in_ready !== 1'b0 || rx_out_valid !== 1'b0 ||
           txd !== 1'b1 || fe !== 1'b0 || ov !== 1'b0)
            $fatal(1,"FAIL RESET_PORTS");
        resetn=1;
        @(negedge clk);
        if(tx_in_ready !== 1'b1) $fatal(1,"FAIL EMPTY_TX_READY");

        // (변경) T1: 16 FIFO + 1 PHY를 초과하는 18개 입력.
        phase=1;
        for(i=0;i<18;i++) push_tx(8'h40+8'(i));
        guard=0;
        while(received<18) begin
            @(negedge clk); guard++;
            if(guard>CPB*10*22) $fatal(1,"FAIL TX_DRAIN_TIMEOUT");
        end
        repeat(CPB*20) begin
            @(negedge clk);
            if(txd !== 1'b1) $fatal(1,"FAIL EXTRA_TX");
        end
        if(accepted!=18 || received!=18 || tx_stalls==0 ||
           tx_full_swap==0 || dut.tx_count !== 5'd0)
            $fatal(1,"FAIL TX_TOTAL_OR_BOUNDARY");
        $display("T1_TX_FIFO_PASS accepted=18 received=18");

        // T2: ready=0으로 RX 16개 저장, 대기 유지 및 순서 검사.
        phase=2;
        for(i=0;i<16;i++) send_rx(8'h80+8'(i));
        repeat(5) @(negedge clk);
        if(dut.rx_count !== 5'd16 || rx_out_valid !== 1'b1 ||
           rx_out_data !== 8'h80 || fe !== 1'b0 || ov !== 1'b0)
            $fatal(1,"FAIL RX_FULL");
        drain_rx(16);
        $display("T2_RX_BACKPRESSURE_PASS checked=16");

        // (추가) T3: FIFO 16 + PHY 보류 1, 18번째 overrun.
        phase=3; rx_checked=0;
        for(i=0;i<18;i++) send_rx(8'ha0+8'(i));
        repeat(5) @(negedge clk);
        if(dut.rx_count !== 5'd16 || dut.phy_rx_valid !== 1'b1 ||
           rx_out_data !== 8'ha0 || fe !== 1'b0 || ov !== 1'b1)
            $fatal(1,"FAIL RX_OVERRUN");
        drain_rx(17);
        if(rx_full_swap==0 || ov !== 1'b1 || fe !== 1'b0)
            $fatal(1,"FAIL RX_SWAP_OR_STICKY");
        $display("T3_RX_OVERRUN_PASS kept=17 dropped=1");

        // (변경) T4: FIFO가 실제로 비어 있지 않을 때 리셋.
        phase=4;
        send_rx(8'hc3);
        repeat(5) @(negedge clk);
        push_tx(8'h00); push_tx(8'h11); push_tx(8'h22);
        if(rx_out_valid !== 1'b1 || rx_out_data !== 8'hc3 ||
           dut.tx_count !== 5'd2 || txd !== 1'b0)
            $fatal(1,"FAIL RESET_PRECONDITION");
        resetn=0;
        repeat(3) @(negedge clk);
        if(rx_out_valid !== 1'b0 || tx_in_ready !== 1'b0 ||
           txd !== 1'b1 || fe !== 1'b0 || ov !== 1'b0)
            $fatal(1,"FAIL RESET_FLUSH");
        resetn=1;
        @(negedge clk);
        if(tx_in_ready !== 1'b1 || rx_out_valid !== 1'b0)
            $fatal(1,"FAIL RESET_RELEASE");
        repeat(CPB*20) begin
            @(negedge clk);
            if(txd !== 1'b1 || rx_out_valid !== 1'b0 ||
               tx_in_ready !== 1'b1)
                $fatal(1,"FAIL RESET_STALE_DATA");
        end
        $display("T4_RESET_PASS");
        if(accepted!=18 || received!=18 || rx_checked!=17)
            $fatal(1,"FAIL FINAL_COUNTS");
        $display("PASS CH14 tx=18 rx=16+17 overrun=1 reset=1");
        $finish;
    end
endmodule
코드역할
push_tx()ready 확인, 상승 에지 수락, 하강 에지 구동
send_rx()직렬 8N1 입력 생성
always @(posedge clk)독립 수락 집계·순서 및 RX 대기 유지
tx_full_swap / rx_full_swapfull 상태에서 동시 교체 발생 확인
decoded[bit_index]직렬 핀의 실제 데이터 직접 복조
drain_rx()RX ready 펄스로 순서대로 소비
RESET_PRECONDITION실제 잔류 데이터가 있는지 확인
GLOBAL_TIMEOUT검증 전체의 무한 대기 방지

이 장의 PASS는 네 시나리오에 한정됩니다. RX 초과 수신은 원본의 의도된 동작이며 모든 보레이트·프레임 오류·리셋 위상을 검증한 것은 아닙니다.

#!/usr/bin/env bash
# 제14장: 원본 uart_core + uart_rx + uart_tx와 TB 컴파일 및 실행.
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
RTL="${RTL_DIR:-$ROOT/reference_rtl}"
if ! command -v iverilog >/dev/null 2>&1 || ! command -v vvp >/dev/null 2>&1; then
  echo "NOT_RUN: iverilog 또는 vvp 없음" >&2
  exit 2
fi
for file in uart_core.v uart_rx.v uart_tx.v; do
  if [[ ! -f "$RTL/$file" ]]; then
    echo "NOT_RUN: 원본 RTL 없음: $RTL/$file" >&2
    exit 2
  fi
done
RTL="$(realpath "$RTL")"
mkdir -p "$ROOT/.build/ch14"
cd "$ROOT/.build/ch14"
iverilog -g2012 -DVCD -s tb_uart_fifo_flow -o sim \
  "$RTL/uart_core.v" "$RTL/uart_rx.v" "$RTL/uart_tx.v" \
  "$ROOT/tb/ch14/tb_uart_fifo_flow.sv"
vvp sim | tee simulation.log
grep -Fxq 'PASS CH14 tx=18 rx=16+17 overrun=1 reset=1' simulation.log
