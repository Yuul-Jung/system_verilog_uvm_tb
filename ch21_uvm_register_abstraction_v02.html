00개요 · 레지스터를 객체로 검증한다

RAL(Register Abstraction Layer, 레지스터 추상화 계층)은 주소·필드·접근 정책·리셋값을 객체로 정의합니다. 제15장의 AXI4-Lite 스냅샷과 제19장의 독립 Monitor 개념을 연결합니다. RAL의 write/read가 실제 버스 거래로 변환되고, 관찰된 거래는 Predictor(프리딕터, 예측값 갱신기)를 통해 Mirror(미러, 모델의 예상 레지스터값)에 반영됩니다.

설계 개념 연계: amd_soc_p2 04.2장 03~04절 AXI·WSTRB, 10.3장 사양서 2~3절 및 03.2장 IRQ 관련 설계. 현행 PWM/IRQ IP와 동일 원본으로 확인되지 않은 범용 스냅샷이므로 개념 연계만 적용합니다.

reference_rtl/axi4lite_slave_adapter.v · Verilog
module axi4lite_slave_adapter #(
    parameter integer C_S_AXI_ADDR_WIDTH = 32
)(
    input  wire                          s_axi_aclk,
    input  wire                          s_axi_aresetn,

    // AXI4-Lite Write Address Channel
    input  wire [C_S_AXI_ADDR_WIDTH-1:0] s_axi_awaddr,
    input  wire [2:0]                    s_axi_awprot,
    input  wire                          s_axi_awvalid,
    output wire                          s_axi_awready,

    // AXI4-Lite Write Data Channel
    input  wire [31:0]                   s_axi_wdata,
    input  wire [3:0]                    s_axi_wstrb,
    input  wire                          s_axi_wvalid,
    output wire                          s_axi_wready,

    // AXI4-Lite Write Response Channel
    output reg  [1:0]                    s_axi_bresp,
    output reg                           s_axi_bvalid,
    input  wire                          s_axi_bready,

    // AXI4-Lite Read Address Channel
    input  wire [C_S_AXI_ADDR_WIDTH-1:0] s_axi_araddr,
    input  wire [2:0]                    s_axi_arprot,
    input  wire                          s_axi_arvalid,
    output wire                          s_axi_arready,

    // AXI4-Lite Read Data Channel
    output reg  [31:0]                   s_axi_rdata,
    output reg  [1:0]                    s_axi_rresp,
    output reg                           s_axi_rvalid,
    input  wire                          s_axi_rready,

    // Bus-independent register access interface
    output wire                          wr_en,
    output wire [7:0]                    wr_offset,
    output wire [31:0]                   wr_data,
    output wire [3:0]                    wr_strb,

    output wire [7:0]                    rd_offset,
    input  wire [31:0]                   rd_data
);

// AXI response code used by this basic slave.
localparam [1:0] AXI_RESP_OKAY = 2'b00;

// Write Address and Write Data are independent AXI channels.
// Each channel is buffered separately until both are available.
reg                          aw_stored;
reg [C_S_AXI_ADDR_WIDTH-1:0] awaddr_stored;
reg                          w_stored;
reg [31:0]                   wdata_stored;
reg [3:0]                    wstrb_stored;

wire write_commit;

// Single-outstanding-write policy:
// Do not accept a new write while a response is pending.
assign s_axi_awready =
    s_axi_aresetn && !aw_stored && !s_axi_bvalid;

assign s_axi_wready =
    s_axi_aresetn && !w_stored && !s_axi_bvalid;

// Register-bank write occurs only after BOTH address and data
// channels have completed their independent handshakes.
assign write_commit =
    aw_stored && w_stored && !s_axi_bvalid;

assign wr_en     = write_commit;
assign wr_offset = awaddr_stored[7:0];
assign wr_data   = wdata_stored;
assign wr_strb   = wstrb_stored;

// Read side accepts one address only when no previous R response
// is waiting for the master.
assign s_axi_arready =
    s_axi_aresetn && !s_axi_rvalid;

// custom_register_bank read path is combinational.
// The address presented during AR handshake selects rd_data.
assign rd_offset = s_axi_araddr[7:0];

// AWPROT / ARPROT are not used by this educational register slave.
wire unused_prot;
assign unused_prot = ^{s_axi_awprot, s_axi_arprot};

always @(posedge s_axi_aclk) begin
    if (!s_axi_aresetn) begin
        aw_stored     <= 1'b0;
        awaddr_stored <= {C_S_AXI_ADDR_WIDTH{1'b0}};
        w_stored      <= 1'b0;
        wdata_stored  <= 32'h0000_0000;
        wstrb_stored  <= 4'b0000;
        s_axi_bresp   <= AXI_RESP_OKAY;
        s_axi_bvalid  <= 1'b0;
    end else begin
        // Capture Write Address independently.
        if (s_axi_awready && s_axi_awvalid) begin
            aw_stored     <= 1'b1;
            awaddr_stored <= s_axi_awaddr;
        end

        // Capture Write Data independently.
        if (s_axi_wready && s_axi_wvalid) begin
            w_stored     <= 1'b1;
            wdata_stored <= s_axi_wdata;
            wstrb_stored <= s_axi_wstrb;
        end

        // Once both parts exist, pulse wr_en through write_commit
        // and create the AXI B response.
        if (write_commit) begin
            aw_stored    <= 1'b0;
            w_stored     <= 1'b0;
            s_axi_bresp  <= AXI_RESP_OKAY;
            s_axi_bvalid <= 1'b1;
        end else if (s_axi_bvalid && s_axi_bready) begin
            s_axi_bvalid <= 1'b0;
        end
    end
end

always @(posedge s_axi_aclk) begin
    if (!s_axi_aresetn) begin
        s_axi_rdata  <= 32'h0000_0000;
        s_axi_rresp  <= AXI_RESP_OKAY;
        s_axi_rvalid <= 1'b0;
    end else begin
        // Capture the selected register value at AR handshake.
        if (s_axi_arready && s_axi_arvalid) begin
            s_axi_rdata  <= rd_data;
            s_axi_rresp  <= AXI_RESP_OKAY;
            s_axi_rvalid <= 1'b1;
        end else if (s_axi_rvalid && s_axi_rready) begin
            s_axi_rvalid <= 1'b0;
        end
    end
end

endmodule
reference_rtl/custom_register_bank.v · Verilog
module custom_register_bank (
    input  wire        clk,
    input  wire        resetn,

    // Bus-independent register access interface.
    // SoC 12 will connect this interface to AXI4-Lite.
    input  wire        wr_en,
    input  wire [7:0]  wr_offset,
    input  wire [31:0] wr_data,
    input  wire [3:0]  wr_strb,

    input  wire [7:0]  rd_offset,
    output reg  [31:0] rd_data,

    // Hardware status / event inputs
    input  wire        busy_i,
    input  wire        done_i,
    input  wire        error_i,
    input  wire        irq_done_event_i,
    input  wire        irq_error_event_i,
    input  wire [31:0] data_out_i,

    // Control outputs
    output reg         enable_o,
    output reg  [1:0]  mode_o,
    output reg         start_pulse_o,
    output reg         clear_pulse_o,
    output reg  [31:0] data_in_o,
    output reg  [1:0]  irq_status_o
);

localparam [7:0] REG_CONTROL    = 8'h00;
localparam [7:0] REG_STATUS     = 8'h04;
localparam [7:0] REG_COMMAND    = 8'h08;
localparam [7:0] REG_IRQ_STATUS = 8'h0C;
localparam [7:0] REG_DATA_IN    = 8'h10;
localparam [7:0] REG_DATA_OUT   = 8'h14;
localparam [7:0] REG_VERSION    = 8'h18;
localparam [7:0] REG_ID         = 8'h1C;

localparam [31:0] VERSION_VALUE = 32'h0001_0000;
localparam [31:0] ID_VALUE      = 32'h524D_4150; // ASCII "RMAP"

integer byte_index;

always @(posedge clk) begin
    if (!resetn) begin
        enable_o      <= 1'b0;
        mode_o        <= 2'b00;
        start_pulse_o <= 1'b0;
        clear_pulse_o <= 1'b0;
        data_in_o     <= 32'h0000_0000;
        irq_status_o  <= 2'b00;
    end else begin
        // W1P pulses are one clock wide by default.
        start_pulse_o <= 1'b0;
        clear_pulse_o <= 1'b0;

        // Hardware events set IRQ status.
        // Event set has priority over software clear in the same cycle.
        if (irq_done_event_i) begin
            irq_status_o[0] <= 1'b1;
        end else if (
            wr_en &&
            (wr_offset == REG_IRQ_STATUS) &&
            wr_strb[0] &&
            wr_data[0]
        ) begin
            irq_status_o[0] <= 1'b0;
        end

        if (irq_error_event_i) begin
            irq_status_o[1] <= 1'b1;
        end else if (
            wr_en &&
            (wr_offset == REG_IRQ_STATUS) &&
            wr_strb[0] &&
            wr_data[1]
        ) begin
            irq_status_o[1] <= 1'b0;
        end

        if (wr_en) begin
            case (wr_offset)
                REG_CONTROL: begin
                    if (wr_strb[0]) begin
                        enable_o <= wr_data[0];
                        mode_o   <= wr_data[2:1];
                    end
                end

                REG_COMMAND: begin
                    if (wr_strb[0]) begin
                        start_pulse_o <= wr_data[0];
                        clear_pulse_o <= wr_data[1];
                    end
                end

                REG_DATA_IN: begin
                    // Byte-enable write using WSTRB-equivalent strobes.
                    for (byte_index = 0; byte_index < 4; byte_index = byte_index + 1) begin
                        if (wr_strb[byte_index]) begin
                            data_in_o[byte_index*8 +: 8] <=
                                wr_data[byte_index*8 +: 8];
                        end
                    end
                end

                default: begin
                    // RO / reserved / unmapped writes have no effect.
                end
            endcase
        end
    end
end

always @(*) begin
    case (rd_offset)
        REG_CONTROL: begin
            rd_data = {29'h0, mode_o, enable_o};
        end

        REG_STATUS: begin
            rd_data = {29'h0, error_i, done_i, busy_i};
        end

        REG_COMMAND: begin
            // W1P command register has no retained state.
            rd_data = 32'h0000_0000;
        end

        REG_IRQ_STATUS: begin
            rd_data = {30'h0, irq_status_o};
        end

        REG_DATA_IN: begin
            rd_data = data_in_o;
        end

        REG_DATA_OUT: begin
            rd_data = data_out_i;
        end

        REG_VERSION: begin
            rd_data = VERSION_VALUE;
        end

        REG_ID: begin
            rd_data = ID_VALUE;
        end

        default: begin
            rd_data = 32'h0000_0000;
        end
    endcase
end

endmodule

어댑터는 AW/W를 따로 저장하고 두 채널이 모두 수락된 다음 bank에 쓰기를 반영합니다. 읽기값은 AR 수락 때 캡처합니다. 이 스냅샷은 미매핑 접근에도 OKAY를 반환하는 교육용 구현이며, 외부 주소 디코딩은 하위 8비트 별칭을 고려해야 합니다.

provides_responses=0이므로 Driver는 요청 객체에 읽기 데이터와 응답을 기록하고 item_done()을 호출합니다. Monitor는 이 객체를 재사용하지 않고 AW/W/B 및 AR/R 수락으로 새 객체를 만들어 발행합니다. 입력은 하강 에지에 구동하며 Monitor는 상승 에지에서 수락된 값을 읽습니다. B/R 수락 후 다음 하강 에지에 Driver가 완료를 통지하므로 Predictor가 먼저 동작하는 구조입니다. 4상태의 X/Z는 2상태 거래 객체로 복사하기 전에 검사합니다.

Mirror는 실제 DUT의 읽기값과 자동으로 같은 값이 아닙니다. 쓰기 직후, 읽기 전에 기대 Mirror를 확인해야 읽기 예측이 쓰기 예측 누락을 가리는 문제를 방지할 수 있습니다.

7개 읽기, 3개 쓰기입니다. 모든 쓰기는 WSTRB=1111로 기대합니다. CONTROL에 FFFFFFFF를 쓰면 실제 하위 3비트만 반영되어 00000007입니다. 이후 5를 쓰면 enable=1, mode=2입니다. VERSION과 ID는 읽기 전용 상수 읽기로 확인합니다. RO 쓰기, W1C, W1P, 외부 이벤트, 부분 쓰기, 모든 AXI 응답 오류는 이번 시험 범위가 아닙니다.

NO_PREDICT는 Monitor와 Predictor의 분석 연결만 생략합니다. 실제 DUT와 Driver는 동일하고 auto_predict도 꺼진 상태입니다. 첫 DATA_IN 쓰기가 끝나도 Mirror가 0에 머물러 PREDICT_WRITE 진단이 발생해야 합니다. 실제 레지스터는 쓰였을 수 있으므로 Mirror 오류와 DUT 오류를 구별합니다. 위 계산은 예상 계약이지 시뮬레이션 로그가 아닙니다.

`timescale 1ns/1ps
// CH21: UVM RAL frontdoor + adapter + predictor. DUT is read-only.
interface axi21_if(input logic clk);
    logic resetn=0;
    logic [31:0] awaddr=0,wdata=0,araddr=0;
    logic [3:0] wstrb=0;
    logic awvalid=0,wvalid=0,bready=0,arvalid=0,rready=0;
    wire awready,wready,bvalid,arready,rvalid;
    wire [1:0] bresp,rresp;
    wire [31:0] rdata;
endinterface

package ch21_pkg;
    import uvm_pkg::*;
    `include "uvm_macros.svh"

    // AXI4-Lite master transaction: one completed read or write.
    class axi21_item extends uvm_sequence_item;
        `uvm_object_utils(axi21_item)
        bit is_write;
        bit [31:0] addr,data;
        bit [3:0] strb=4'hf;
        bit [1:0] resp;
        function new(string name="axi21_item");
            super.new(name);
        endfunction
    endclass

    class axi21_adapter extends uvm_reg_adapter;
        `uvm_object_utils(axi21_adapter)
        function new(string name="axi21_adapter");
            super.new(name);
            // (변경) 전체 레지스터 접근만 수행한다. 개별 필드 byte 접근은 미지원.
            supports_byte_enable=0;
            provides_responses=0;
        endfunction
        virtual function uvm_sequence_item reg2bus(const ref uvm_reg_bus_op rw);
            axi21_item t=axi21_item::type_id::create("t");
            t.is_write=(rw.kind==UVM_WRITE);
            t.addr=rw.addr;
            t.data=rw.data;
            t.strb=rw.byte_en;
            return t;
        endfunction
        virtual function void bus2reg(uvm_sequence_item bus_item,
                                       ref uvm_reg_bus_op rw);
            axi21_item t;
            if (!$cast(t,bus_item))
                `uvm_fatal("ADAPTER","Unexpected bus transaction")
            rw.kind=t.is_write ? UVM_WRITE : UVM_READ;
            rw.addr=t.addr;
            rw.data=t.data;
            rw.byte_en=t.strb;
            rw.status=(t.resp==2'b00) ? UVM_IS_OK : UVM_NOT_OK;
        endfunction
    endclass

    class axi21_reg extends uvm_reg;
        `uvm_object_utils(axi21_reg)
        uvm_reg_field value;
        string access_mode="RW";
        bit is_control=0;
        uvm_reg_field mode, reserved;
        uvm_reg_data_t reset_value=0;
        function new(string name="axi21_reg");
            super.new(name,32,UVM_NO_COVERAGE);
        endfunction
        virtual function void build();
            value=uvm_reg_field::type_id::create("value");
            // (변경) CONTROL의 실제 쓰기 가능 범위는 3비트뿐이다.
            if (is_control) begin
                value.configure(this,1,0,"RW",0,0,1,0,0); // enable
                mode=uvm_reg_field::type_id::create("mode");
                mode.configure(this,2,1,"RW",0,0,1,0,0);
                reserved=uvm_reg_field::type_id::create("reserved");
                reserved.configure(this,29,3,"RO",0,0,1,0,0);
            end else
                value.configure(this,32,0,access_mode,0,reset_value,1,0,0);
        endfunction
    endclass

    class axi21_block extends uvm_reg_block;
        `uvm_object_utils(axi21_block)
        axi21_reg control,data_in,version_reg,id_reg;
        function new(string name="axi21_block");
            super.new(name,UVM_NO_COVERAGE);
        endfunction
        virtual function void build();
            default_map=create_map("map",0,4,UVM_LITTLE_ENDIAN,1);
            control=axi21_reg::type_id::create("control");
            control.is_control=1;
            control.configure(this);
            control.build();
            default_map.add_reg(control,'h00,"RW");
            data_in=axi21_reg::type_id::create("data_in");
            data_in.configure(this);
            data_in.build();
            default_map.add_reg(data_in,'h10,"RW");
            version_reg=axi21_reg::type_id::create("version_reg");
            version_reg.access_mode="RO";
            version_reg.reset_value=32'h00010000;
            version_reg.configure(this);
            version_reg.build();
            default_map.add_reg(version_reg,'h18,"RO");
            id_reg=axi21_reg::type_id::create("id_reg");
            id_reg.access_mode="RO";
            id_reg.reset_value=32'h524D4150;
            id_reg.configure(this);
            id_reg.build();
            default_map.add_reg(id_reg,'h1c,"RO");
            lock_model();
        endfunction
    endclass

    class axi21_sequencer extends uvm_sequencer #(axi21_item);
        `uvm_component_utils(axi21_sequencer)
        function new(string name,uvm_component parent);
            super.new(name,parent);
        endfunction
    endclass

    class axi21_driver extends uvm_driver #(axi21_item);
        `uvm_component_utils(axi21_driver)
        virtual axi21_if vif;
        function new(string name,uvm_component parent);
            super.new(name,parent);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual axi21_if)::get(this,"","vif",vif))
                `uvm_fatal("NO_VIF","Missing AXI interface")
        endfunction
        task automatic write_one(axi21_item t);
            // AW and W independent; hold both until their handshakes.
            bit aw_done=0,w_done=0;
            @(negedge vif.clk);
            vif.awaddr=t.addr;vif.wdata=t.data;vif.wstrb=t.strb;
            vif.awvalid=1;vif.wvalid=1;
            for(int k=0;k<100;k++) begin
                @(posedge vif.clk);
                if(vif.awvalid && vif.awready) aw_done=1;
                if(vif.wvalid && vif.wready) w_done=1;
                @(negedge vif.clk);
                if(aw_done) vif.awvalid=0;
                if(w_done) vif.wvalid=0;
                if(aw_done && w_done) break;
            end
            if(!aw_done || !w_done) `uvm_fatal("AW_W","Handshake timeout")
            for(int k=0;k<100;k++) begin
                @(negedge vif.clk);
                if(vif.bvalid) begin
                    t.resp=vif.bresp;
                    vif.bready=1;
                    @(posedge vif.clk);
                    if (vif.bvalid !== 1'b1 || vif.bresp !== t.resp)
                        `uvm_fatal("B_HANDSHAKE","Response changed before acceptance")
                    @(negedge vif.clk);
                    vif.bready=0;
                    return;
                end
            end
            `uvm_fatal("B_TIMEOUT","Write response timeout")
        endtask
        task automatic read_one(axi21_item t);
            bit accepted=0;
            @(negedge vif.clk);
            vif.araddr=t.addr;vif.arvalid=1;
            for(int k=0;k<100;k++) begin
                @(posedge vif.clk);
                if(vif.arvalid && vif.arready) accepted=1;
                @(negedge vif.clk);
                if(accepted) begin vif.arvalid=0;break;end
            end
            if(!accepted) `uvm_fatal("AR_TIMEOUT","Address timeout")
            for(int k=0;k<100;k++) begin
                @(negedge vif.clk);
                if(vif.rvalid) begin
                    t.data=vif.rdata;t.resp=vif.rresp;
                    vif.rready=1;
                    @(posedge vif.clk);
                    if (vif.rvalid !== 1'b1 || vif.rresp !== t.resp ||
                        vif.rdata !== t.data)
                        `uvm_fatal("R_HANDSHAKE","Response changed before acceptance")
                    @(negedge vif.clk);
                    vif.rready=0;
                    return;
                end
            end
            `uvm_fatal("R_TIMEOUT","Read response timeout")
        endtask
        task run_phase(uvm_phase phase);
            axi21_item req;
            forever begin
                seq_item_port.get_next_item(req);
                if(req.is_write) write_one(req);
                else read_one(req);
                seq_item_port.item_done();
            end
        endtask
    endclass

    // Driver의 요청 객체를 Predictor로 전달하지 않는다.
    class axi21_monitor extends uvm_monitor;
        `uvm_component_utils(axi21_monitor)
        virtual axi21_if vif;
        uvm_analysis_port #(axi21_item) ap;
        bit have_aw=0, have_w=0, have_ar=0;
        logic [31:0] wa, wd, ra;
        logic [3:0] ws;
        int unsigned writes=0, reads=0;
        axi21_item last_write, last_read;
        function new(string name,uvm_component parent);
            super.new(name,parent);
            ap=new("ap",this);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if(!uvm_config_db#(virtual axi21_if)::get(this,"","vif",vif))
                `uvm_fatal("NO_VIF","Monitor interface missing")
        endfunction
        task run_phase(uvm_phase phase);
            axi21_item t;
            forever begin
                @(posedge vif.clk);
                if(vif.resetn === 1'b0) begin
                    have_aw=0; have_w=0; have_ar=0;
                end else begin
                    if ((^{vif.resetn,vif.awvalid,vif.awready,
                           vif.wvalid,vif.wready,vif.bvalid,vif.bready,
                           vif.arvalid,vif.arready,vif.rvalid,vif.rready}) === 1'bx)
                        `uvm_fatal("BUS_X","Unknown handshake signal")
                    if(vif.awvalid && vif.awready) begin
                        if(have_aw || (^vif.awaddr) === 1'bx)
                            `uvm_fatal("MON_AW","Duplicate or unknown AW")
                        wa=vif.awaddr; have_aw=1;
                    end
                    if(vif.wvalid && vif.wready) begin
                        if(have_w || (^{vif.wdata,vif.wstrb}) === 1'bx)
                            `uvm_fatal("MON_W","Duplicate or unknown W")
                        wd=vif.wdata; ws=vif.wstrb; have_w=1;
                    end
                    if(vif.bvalid && vif.bready) begin
                        if(!have_aw || !have_w || vif.bresp !== 2'b00)
                            `uvm_fatal("MON_B","Unexpected write response")
                        t=axi21_item::type_id::create("observed_write");
                        t.is_write=1; t.addr=wa; t.data=wd; t.strb=ws;
                        t.resp=vif.bresp;
                        have_aw=0; have_w=0;
                        writes++; last_write=t; ap.write(t);
                    end
                    if(vif.arvalid && vif.arready) begin
                        if(have_ar || (^vif.araddr) === 1'bx)
                            `uvm_fatal("MON_AR","Duplicate or unknown AR")
                        ra=vif.araddr; have_ar=1;
                    end
                    if(vif.rvalid && vif.rready) begin
                        if(!have_ar || vif.rresp !== 2'b00 ||
                           (^vif.rdata) === 1'bx)
                            `uvm_fatal("MON_R","Unexpected read response")
                        t=axi21_item::type_id::create("observed_read");
                        t.is_write=0; t.addr=ra; t.data=vif.rdata;
                        t.strb=4'hf; t.resp=vif.rresp;
                        have_ar=0;
                        reads++; last_read=t; ap.write(t);
                    end
                end
            end
        endtask
    endclass

    class axi21_test extends uvm_test;
        `uvm_component_utils(axi21_test)
        axi21_env env;
        virtual axi21_if vif;
        int unsigned read_checks=0, write_checks=0;
        bit done=0;
        function new(string name,uvm_component parent);
            super.new(name,parent);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            env=axi21_env::type_id::create("env",this);
            if(!uvm_config_db#(virtual axi21_if)::get(this,"","vif",vif))
                `uvm_fatal("NO_VIF","Test interface missing")
        endfunction
        // (추가) RAL 반환과 실제 버스 거래를 함께 확인한다.
        task automatic write_expect(uvm_reg rg, bit [31:0] addr,
                                    bit [31:0] data, bit [31:0] expected);
            uvm_status_e status;
            rg.write(status,data,UVM_FRONTDOOR,env.model.default_map);
            write_checks++;
            if(status!=UVM_IS_OK || env.mon.writes!=write_checks ||
               env.mon.last_write==null ||
               env.mon.last_write.addr!==addr ||
               env.mon.last_write.data!==data ||
               env.mon.last_write.strb!==4'hf)
                `uvm_fatal("WRITE","Write status or observed transaction mismatch")
            // 읽기 전에 확인해야 read prediction이 쓰기 예측 누락을 가리지 않는다.
            if(rg.get_mirrored_value() !== uvm_reg_data_t'(expected))
                `uvm_fatal("PREDICT_WRITE",$sformatf(
                    "addr=%h expected=%h",addr,expected))
        endtask

        task automatic read_expect(uvm_reg rg, bit [31:0] addr,
                                   bit [31:0] expected);
            uvm_status_e status;
            uvm_reg_data_t value;
            rg.read(status,value,UVM_FRONTDOOR,env.model.default_map);
            read_checks++;
            if(status!=UVM_IS_OK || value!==uvm_reg_data_t'(expected) ||
               env.mon.reads!=read_checks || env.mon.last_read==null ||
               env.mon.last_read.addr!==addr ||
               env.mon.last_read.data!==expected ||
               rg.get_mirrored_value()!==uvm_reg_data_t'(expected))
                `uvm_fatal("READ",$sformatf("Read mismatch addr=%h",addr))
        endtask

        task run_phase(uvm_phase phase);
            phase.raise_objection(this);
            repeat(4) @(negedge vif.clk);
            vif.resetn=1;
            @(negedge vif.clk);
            env.model.reset(); // 모델만 초기화한다. DUT reset은 위에서 수행.

            read_expect(env.model.control,32'h00,32'h0);
            read_expect(env.model.data_in,32'h10,32'h0);
            write_expect(env.model.data_in,32'h10,32'h11223344,32'h11223344);
            read_expect(env.model.data_in,32'h10,32'h11223344);
            // (추가) 예약 비트는 1을 써도 읽기값과 mirror가 0이어야 한다.
            write_expect(env.model.control,32'h00,32'hFFFFFFFF,32'h7);
            read_expect(env.model.control,32'h00,32'h7);
            write_expect(env.model.control,32'h00,32'h5,32'h5);
            read_expect(env.model.control,32'h00,32'h5);
            read_expect(env.model.version_reg,32'h18,32'h00010000);
            read_expect(env.model.id_reg,32'h1C,32'h524D4150);
            done=1;
            phase.drop_objection(this);
        endtask
        function void check_phase(uvm_phase phase);
            super.check_phase(phase);
            if(!done || read_checks!=7 || write_checks!=3 ||
               env.mon.reads!=7 || env.mon.writes!=3 ||
               env.mon.have_aw || env.mon.have_w || env.mon.have_ar)
                `uvm_fatal("COUNTS","Incomplete RAL verification")
        endfunction
        function void report_phase(uvm_phase phase);
            super.report_phase(phase);
            if (uvm_report_server::get_server().get_severity_count(UVM_ERROR)!=0 ||
                uvm_report_server::get_server().get_severity_count(UVM_FATAL)!=0)
                `uvm_fatal("REPORT","UVM error or fatal reported")
            `uvm_info("RESULT","PASS CH21 RAL reads=7 writes=3",UVM_NONE)
        endfunction
    endclass
endpackage

#!/usr/bin/env bash
# 제21장: Vivado XSIM 내장 UVM 라이브러리 사용.
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
RTL="${RTL_DIR:-$ROOT/reference_rtl}"
for f in axi4lite_slave_adapter.v custom_register_bank.v; do
  if [[ ! -f "$RTL/$f" ]]; then echo "NOT_RUN: missing $RTL/$f" >&2; exit 2; fi
done
RTL="$(realpath "$RTL")"
for tool in xvlog xelab xsim; do
  if ! command -v "$tool" >/dev/null 2>&1; then
    echo "NOT_RUN: $tool missing; source Vivado settings64.sh" >&2
    exit 2
  fi
done
mkdir -p "$ROOT/.build/ch21"
OUT="$(mktemp -d "$ROOT/.build/ch21/run.XXXXXX")"
cd "$OUT"
xvlog "$RTL/axi4lite_slave_adapter.v" "$RTL/custom_register_bank.v" 2>&1 | tee rtl.log
xvlog -sv -L uvm "$ROOT/tb/ch21/tb_axi_uvm_ral.sv" 2>&1 | tee tb.log
xelab -L uvm -debug typical tb_axi_uvm_ral -s ch21_sim 2>&1 | tee elaborate.log
xsim ch21_sim -runall -onerror quit 2>&1 | tee normal.log
grep -Eq '\[RESULT\][[:space:]]+PASS CH21 RAL reads=7 writes=3[[:space:]]*$' normal.log
grep -Eq 'UVM_ERROR[[:space:]]*:[[:space:]]*0[[:space:]]*$' normal.log
grep -Eq 'UVM_FATAL[[:space:]]*:[[:space:]]*0[[:space:]]*$' normal.log
rc=0
xsim ch21_sim -runall -onerror quit -testplusarg NO_PREDICT > negative.log 2>&1 || rc=$?
grep -Eq 'UVM_FATAL.*\[PREDICT_WRITE\].*addr=00000010 expected=11223344' negative.log
if grep -Eq 'PASS CH21|CH21 GLOBAL_TIMEOUT' negative.log; then
  echo "FAIL: unexpected negative result" >&2
  exit 1
fi
echo "PASS CH21 HARNESS (negative exit=$rc)"
echo "Logs: $OUT"
