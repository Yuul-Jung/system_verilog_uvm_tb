01원본 RTL과 PWM 출력 계약

RESET_PERIOD와 cfg_period는 8클록으로 고정합니다. 클록은 10ns, 한 PWM 주기는 80ns입니다. 대기 설정은 주기 경계에서 활성 설정으로 이동합니다. 리셋은 레지스터에 동기적으로 적용되며 출력은 조합적으로 차단됩니다.

reference_rtl/pwm_core.v · Verilog
`timescale 1ns/1ps
// P2-04: 보드와 무관한 단일 채널 범용 PWM 코어.
// (추가) period와 duty는 클록 횟수 단위. 0%/100%는 지속 LOW/HIGH.
// (추가) 설정은 주기 마지막 경계에서만 활성값으로 반영.
module pwm_core #(
    parameter [31:0] RESET_PERIOD = 32'd100000
)(
    input  wire        clk,
    input  wire        resetn,
    input  wire        cfg_apply,
    input  wire [31:0] cfg_period,
    input  wire [31:0] cfg_duty,
    input  wire        cfg_enable,
    input  wire        cfg_polarity,
    output wire        pwm_out,
    output reg         pending,
    output reg [31:0] active_period,
    output reg [31:0] active_duty,
    output reg         active_enable,
    output reg         active_polarity
);
    reg [31:0] counter;
    reg [31:0] shadow_period;
    reg [31:0] shadow_duty;
    reg        shadow_enable;
    reg        shadow_polarity;

    wire boundary = (counter >= active_period - 32'd1);
    wire duty_high = (counter < active_duty);

    // (주의) enable=0이면 polarity와 관계없이 LOW로 차단.
    // (설명) 기본 polarity에서 active_duty=0이면 항상 LOW,
    //         active_duty=active_period이면 항상 HIGH.
    assign pwm_out = resetn && active_enable &&
                     (active_polarity ? !duty_high : duty_high);

    always @(posedge clk) begin
        if (!resetn) begin
            counter         <= 32'd0;
            active_period   <= RESET_PERIOD;
            active_duty     <= 32'd0;
            active_enable   <= 1'b0;
            active_polarity <= 1'b0;
            shadow_period   <= RESET_PERIOD;
            shadow_duty     <= 32'd0;
            shadow_enable   <= 1'b0;
            shadow_polarity <= 1'b0;
            pending         <= 1'b0;
        end else begin
            if (boundary) begin
                counter <= 32'd0;
                if (pending) begin
                    active_period   <= shadow_period;
                    active_duty     <= shadow_duty;
                    active_enable   <= shadow_enable;
                    active_polarity <= shadow_polarity;
                    pending         <= 1'b0;
                end
            end else begin
                counter <= counter + 32'd1;
            end
            // (추가) 경계와 APPLY가 겹치면 새 값은 다음 경계에 적용.
            // 여러 APPLY가 겹치면 최종 요청값을 우선하는 단일 Shadow 구조.
            if (cfg_apply) begin
                shadow_period   <= cfg_period;
                shadow_duty     <= cfg_duty;
                shadow_enable   <= cfg_enable;
                shadow_polarity <= cfg_polarity;
                pending         <= 1'b1;
            end
        end
    end
endmodule

정상 극성의 HIGH 개수는 듀티, 반전 극성은 8에서 듀티를 뺀 값입니다. enable이 0이면 극성과 무관하게 LOW입니다. 이번 장에서는 공개 포트만 관찰합니다.

02Covergroup·Coverpoint·Bins

Covergroup(커버그룹, 검증 항목 묶음)은 샘플의 범위를 기록합니다. Coverpoint(커버포인트, 관찰 항목)는 값을 분류하고 bin(빈, 값의 분류 구간)은 방문해야 할 범위를 나타냅니다.

커버포인트bin범위
cp_dutyzero / middle / full0 / 1~7 / 8
cp_enabledisabled / enabled0 / 1
cp_polaritynormal / inverted0 / 1

bins middle = {[1:7]};은 1~7을 하나의 bin으로 묶습니다. 이번 시험은 대표값 4만 방문하므로 1·2·3·5·6·7을 시험하지 않습니다. 모든 값을 별도 bin으로 나누는 bins middle[] = {[1:7]};과는 다릅니다. illegal_bins outside = default;는 정상 듀티 범위 외 값을 불법으로 분류합니다. 이 실습은 범위 밖 설정을 실제로 주입하지 않습니다.

03Cross Coverage · 교차 조합 12개

Cross Coverage(크로스 커버리지, 여러 항목의 교차 범위)는 듀티 구간 3개, enable 2개, polarity 2개를 교차합니다. 각각의 값이 모두 관찰되어도 모든 조합을 시험했다는 뜻은 아닙니다.

전체 합법 입력값을 개별 값으로 세면 9 × 2 × 2 → 36개 조합입니다. 현재 12개 bin의 대표 시험만 실행합니다. get_coverage()는 한 개 생성된 covergroup의 타입 종합 비율이며, 교차 조합의 정확한 방문은 별도의 visited[11:0]로 확인합니다. 12개 비트가 모두 1이면 12개 분류가 시험됐음을 뜻합니다.

04검증 절차 · 적용 확인 후 파형과 커버리지

run_case()는 하강 에지에서 설정 요청 → pending 해제 및 활성 설정 확인 → 8개 출력 표본 검사 → 커버리지 샘플 → visited 집계 순서로 실행됩니다. 적용을 관찰한 하강 에지는 위상 0이며, 이어지는 표본은 위상 1, 2, …, 7, 0입니다. 공개 포트의 적용 완료 시점을 기준으로 추론하며 내부 counter를 읽지 않습니다.

예를 들어 듀티 4, 정상 극성, enable 1의 기대 표본은 11100001입니다. 같은 HIGH 개수라도 순서가 다르면 기존 개수 비교만으로는 검출하지 못하므로 개별 표본의 기대 레벨도 비교합니다. 활성 설정과 pending이 각 표본에서 유지되는지도 확인합니다. 단, 잘못된 적용 시점과 출력이 함께 일관되게 이동하는 일부 결함은 놓칠 수 있습니다. 샘플 사이 순간 글리치, 임의 주기와 연속 요청 전체는 검사 범위 밖입니다.

05테스트벤치 전체 코드

tb/ch12/tb_pwm_coverage.sv · SystemVerilog
`timescale 1ns/1ps
// 제12장: 원본 pwm_core의 설정 조합을 기능 커버리지로 측정한다.
// 기능 커버리지는 '입력 조합을 실제로 적용했는가'를 기록한다.
// PWM의 정확한 파형은 8클록 HIGH 개수 검사로 별도 판정한다.
module tb_pwm_coverage;
    logic clk = 0;
    logic resetn = 0;
    logic cfg_apply = 0;
    logic [31:0] cfg_period = 32'd8;
    logic [31:0] cfg_duty = 0;
    logic cfg_enable = 0;
    logic cfg_polarity = 0;
    wire pwm_out, pending, active_enable, active_polarity;
    wire [31:0] active_period, active_duty;
    int checks = 0;
    // (추가) 커버리지 엔진과 별개로 12개 교차 조합을 집계한다.
    bit [11:0] visited = 12'b0;

    pwm_core #(.RESET_PERIOD(32'd8)) dut (
        .clk(clk), .resetn(resetn), .cfg_apply(cfg_apply),
        .cfg_period(cfg_period), .cfg_duty(cfg_duty),
        .cfg_enable(cfg_enable), .cfg_polarity(cfg_polarity),
        .pwm_out(pwm_out), .pending(pending),
        .active_period(active_period), .active_duty(active_duty),
        .active_enable(active_enable), .active_polarity(active_polarity)
    );
    always #5 clk = ~clk;

    // 명시적 sample()을 호출하므로 리셋/설정 대기 구간은 기록하지 않는다.
    covergroup pwm_cfg_cg with function sample(
        int unsigned duty, bit en, bit pol
    );
        option.per_instance = 1;
        cp_duty: coverpoint duty {
            bins zero = {0};
            bins middle = {[1:7]};
            bins full = {8};
            illegal_bins outside = default;
        }
        cp_enable: coverpoint en {
            bins disabled = {0};
            bins enabled = {1};
        }
        cp_polarity: coverpoint pol {
            bins normal = {0};
            bins inverted = {1};
        }
        cx_settings: cross cp_duty, cp_enable, cp_polarity;
    endgroup
    pwm_cfg_cg cov = new();

    task automatic run_case(
        input logic [31:0] duty,
        input logic en, input logic pol
    );
        int waited;
        int highs;
        int expected_highs;
        int sample_phase, bin_index;
        logic expected_level;
        // 음의 에지에서 입력을 바꿔 양의 에지 샘플과 충돌하지 않는다.
        @(negedge clk);
        cfg_duty = duty;
        cfg_enable = en;
        cfg_polarity = pol;
        cfg_apply = 1;
        @(negedge clk);
        cfg_apply = 0;
        if (pending !== 1'b1)
            $fatal(1, "FAIL pending did not assert");

        // 실제 DUT는 shadow 값을 주기 경계에서 활성화한다.
        waited = 0;
        while (pending !== 1'b0 && waited < 24) begin
            @(negedge clk);
            waited++;
        end
        if (pending !== 1'b0 || active_period !== 32'd8 ||
            active_duty !== duty || active_enable !== en ||
            active_polarity !== pol)
            $fatal(1, "FAIL apply duty=%0d en=%b pol=%b", duty, en, pol);

        // 8개의 연속된 PWM 클록을 관찰하면 정확히 한 주기이다.
        // enable=0이면 polarity와 무관하게 LOW가 유지된다.
        highs = 0;
        for (int i = 0; i < 8; i++) begin
            @(negedge clk);
            if (pwm_out === 1'bx || pwm_out === 1'bz)
                $fatal(1, "FAIL unknown pwm_out");
            // (추가) 적용 관찰 직후 다음 하강 에지는 phase 1,
            // 마지막 표본은 다음 주기의 phase 0이다.
            sample_phase = (i + 1) % 8;
            expected_level = en && (pol ? (sample_phase >= int'(duty))
                                        : (sample_phase < int'(duty)));
            if (pwm_out !== expected_level)
                $fatal(1, "FAIL phase duty=%0d phase=%0d actual=%b expected=%b",
                       duty, sample_phase, pwm_out, expected_level);
            if (active_period !== 32'd8 || active_duty !== duty ||
                active_enable !== en || active_polarity !== pol ||
                pending !== 1'b0)
                $fatal(1, "FAIL active state changed during sampling");
            if (pwm_out) highs++;
        end
        expected_highs = en ? (pol ? 8 - int'(duty) : int'(duty)) : 0;
        if (highs != expected_highs)
            $fatal(1, "FAIL waveform duty=%0d en=%b pol=%b high=%0d expected=%0d",
                   duty, en, pol, highs, expected_highs);

        // 출력 검사가 끝난 실제 적용 조합만 커버리지에 포함한다.
        // (변경) 확인한 DUT의 실제 활성 설정으로 샘플링한다.
        cov.sample(active_duty, active_enable, active_polarity);
        // (추가) zero/middle/full → 0/1/2, 각 구간에 4개 조합.
        bin_index = ((duty == 0) ? 0 : (duty == 8) ? 2 : 1) * 4
                    + int'(en) * 2 + int'(pol);
        visited[bin_index] = 1'b1;
        checks++;
        $display("CHECK duty=%0d en=%b pol=%b highs=%0d",
                 duty, en, pol, highs);
    endtask

    initial begin
        #10000;
        $fatal(1, "FAIL GLOBAL_TIMEOUT");
    end

`ifdef VCD
    // (변경) VCD가 지원되는 실행 환경에서만 활성화한다.
    initial begin
        $dumpfile("ch12_pwm_coverage.vcd");
        $dumpvars(0, tb_pwm_coverage);
    end
`endif
    initial begin
        repeat (3) @(negedge clk);
        resetn = 1;
        // 3 듀티 구간 × 2 enable × 2 polarity = 12개 cross bin.
        for (int d = 0; d < 3; d++) begin
            for (int e = 0; e < 2; e++) begin
                for (int p = 0; p < 2; p++) begin
                    case (d)
                        0: run_case(32'd0, bit'(e), bit'(p));
                        1: run_case(32'd4, bit'(e), bit'(p));
                        2: run_case(32'd8, bit'(e), bit'(p));
                    endcase
                end
            end
        end
        if (checks != 12 || visited !== 12'hfff)
            $fatal(1, "FAIL missing test cases visited=%03h", visited);
        if (cov.get_coverage() < 99.99)
            $fatal(1, "FAIL coverage=%0.2f", cov.get_coverage());
        $display("PASS CH12 cases=%0d coverage=%0.2f%%",
                 checks, cov.get_coverage());
        $finish;
    end
endmodule

핵심 코드 풀이

코드역할

12개 사례 각각 8회 표본 비교를 수행하므로 정상 경로는 96개 출력 표본을 검증합니다. checks는 표본 수가 아닌 완료 사례 수입니다. 모든 시험을 통과한 후에만 최종 PASS를 출력합니다.

06Ubuntu 실행 · Vivado XSIM과 커버리지 보고서

AMD UG900의 SystemVerilog 지원표(2024.1)는 covergroup, coverpoint, cross, 사용자 sample 및 사전 정의 커버리지 메서드를 지원한다고 명시합니다. 설치 버전에 따라 실제 지원 여부를 확인합니다. Icarus에서 이 covergroup 실습 전체를 지원한다고 가정하지 않습니다.

tb/ch12/run_ch12.sh · Bash
#!/usr/bin/env bash
# 제12장: 독립 PWM 원본 + XSIM 기능 커버리지 시험.
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/../.." && pwd)"
RTL="${1:-$ROOT/reference_rtl/pwm_core.v}"
if [[ ! -f "$RTL" ]]; then
  echo "FAIL: pwm_core.v 파일 없음" >&2
  exit 1
fi
RTL="$(realpath "$RTL")"
for tool in xvlog xelab xsim; do
  if ! command -v "$tool" >/dev/null 2>&1; then
    echo "NOT_RUN: $tool 없음. Vivado 환경을 설정하세요." >&2
    exit 2
  fi
done
mkdir -p "$ROOT/.build/ch12"
OUT="$(mktemp -d "$ROOT/.build/ch12/run.XXXXXX")"
cd "$OUT"
xvlog "$RTL" 2>&1 | tee rtl.log
xvlog -sv "$ROOT/tb/ch12/tb_pwm_coverage.sv" 2>&1 | tee tb.log
xelab -debug typical -cov_db_dir . -cov_db_name ch12_cov \
  tb_pwm_coverage -s ch12_sim 2>&1 | tee elaborate.log
xsim ch12_sim -runall -cov_db_dir . -cov_db_name ch12_cov \
  2>&1 | tee output.log
grep -Eq '^PASS CH12 cases=12 coverage=100\.00%[[:space:]]*$' output.log
if command -v xcrg >/dev/null 2>&1; then
  xcrg -dir . -db_name ch12_cov -report_dir report -report_format html
else
  echo "NOT_RUN: xcrg 보고서 생성 도구 없음"
fi
echo "실행 폴더: $OUT"
