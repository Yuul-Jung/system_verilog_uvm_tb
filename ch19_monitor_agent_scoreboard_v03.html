00개요 · 구동·관찰·판정 분리

제18장은 Driver 내부에서 PWM을 비교했습니다. 제19장은 Driver(드라이버, 입력 구동기), Monitor(모니터, 독립 관찰기), Scoreboard(스코어보드, 기대값 비교기)를 분리합니다. Agent(에이전트, 인터페이스 검증 구성 단위)는 Sequencer·Driver·Monitor를 묶습니다.

선행 설계 학습안내서: amd_soc_p2 04.2장 RTL·AXI 풀이 02절 「설정 Shadow와 출력 Active를 분리」 및 06절 「독립 PWM Core 전체」. 원본 PWM을 사용하며 AXI와 보드는 범위 밖입니다.

01원본 PWM RTL 전체

10ns 클록, RESET_PERIOD=8입니다. cfg_apply는 상승 에지에서 Shadow를 저장하고 주기 경계에서 Active로 반영합니다. 리셋은 내부 상태에는 동기식이며 출력은 조합적으로 차단됩니다.

reference_rtl/pwm_core.v · Verilog
`timescale 1ns/1ps
// P2-04: 보드와 무관한 단일 채널 범용 PWM 코어.
// (추가) period와 duty는 클록 횟수 단위. 0%/100%는 지속 LOW/HIGH.
// (추가) 설정은 주기 마지막 경계에서만 활성값으로 반영.
module pwm_core #(
    parameter [31:0] RESET_PERIOD = 32'd100000
)(
    input  wire        clk,
    input  wire        resetn,
    input  wire        cfg_apply,
    input  wire [31:0] cfg_period,
    input  wire [31:0] cfg_duty,
    input  wire        cfg_enable,
    input  wire        cfg_polarity,
    output wire        pwm_out,
    output reg         pending,
    output reg [31:0] active_period,
    output reg [31:0] active_duty,
    output reg         active_enable,
    output reg         active_polarity
);
    reg [31:0] counter;
    reg [31:0] shadow_period;
    reg [31:0] shadow_duty;
    reg        shadow_enable;
    reg        shadow_polarity;

    wire boundary = (counter >= active_period - 32'd1);
    wire duty_high = (counter < active_duty);

    // (주의) enable=0이면 polarity와 관계없이 LOW로 차단.
    // (설명) 기본 polarity에서 active_duty=0이면 항상 LOW,
    //         active_duty=active_period이면 항상 HIGH.
    assign pwm_out = resetn && active_enable &&
                     (active_polarity ? !duty_high : duty_high);

    always @(posedge clk) begin
        if (!resetn) begin
            counter         <= 32'd0;
            active_period   <= RESET_PERIOD;
            active_duty     <= 32'd0;
            active_enable   <= 1'b0;
            active_polarity <= 1'b0;
            shadow_period   <= RESET_PERIOD;
            shadow_duty     <= 32'd0;
            shadow_enable   <= 1'b0;
            shadow_polarity <= 1'b0;
            pending         <= 1'b0;
        end else begin
            if (boundary) begin
                counter <= 32'd0;
                if (pending) begin
                    active_period   <= shadow_period;
                    active_duty     <= shadow_duty;
                    active_enable   <= shadow_enable;
                    active_polarity <= shadow_polarity;
                    pending         <= 1'b0;
                end
            end else begin
                counter <= counter + 32'd1;
            end
            // (추가) 경계와 APPLY가 겹치면 새 값은 다음 경계에 적용.
            // 여러 APPLY가 겹치면 최종 요청값을 우선하는 단일 Shadow 구조.
            if (cfg_apply) begin
                shadow_period   <= cfg_period;
                shadow_duty     <= cfg_duty;
                shadow_enable   <= cfg_enable;
                shadow_polarity <= cfg_polarity;
                pending         <= 1'b1;
            end
        end
    end
endmodule

기본 극성·주기 8·듀티 2, 6, 0의 순차 요청을 검증합니다. DUT의 내부 counter와 Shadow 레지스터는 읽지 않습니다.

Agent는 항상 능동형이며 수동형 전환 기능은 없습니다. drv.seq_item_port.connect(sqr.seq_item_export)는 요청/완료의 연결입니다. ag.mon.ap.connect(sb.analysis_export)는 관찰 전달 연결입니다. Analysis Port(애널리시스 포트, 분석 데이터 발행 포트)의 write는 시간을 소비하지 않는 함수이며 Analysis Imp(애널리시스 임프, 수신 구현 포트)가 sb.write를 호출합니다. 매 클록 새 객체를 생성하므로 이전 관찰값과 객체 참조가 뒤섞이지 않습니다.

03독립 기대 모델 · 매 클록 비교

Driver는 하강 에지에서 입력을 구동합니다. Monitor는 상승 에지 1ns 뒤 비차단 대입이 반영된 공개 신호를 수집합니다. pending 변화만 관찰하면 잘못된 pending 때문에 표본 자체가 누락될 수 있습니다. 따라서 매 에지의 원시 신호를 전달합니다.

Scoreboard 단계동작
초기 3개 리셋 표본공개 출력 6개 초기값 비교
cycles 증가리셋 후 클록 집계
phase_idx = cycles % 8독립 위상 계산
phase_idx == 0이전 요청의 예상 적용
cfg_apply 관찰독립 planned_duty와 요청 비교
매 에지active_* 4개, pending, pwm 비교
samples 증가예상 활성 상태 24개 표본 집계

기대 모델은 고정 주기 8에서 유효합니다. 경계와 요청이 겹치면 이전 요청을 먼저 처리하고 새 요청을 저장합니다. DUT의 pending이나 active_duty에 맞춰 기대 시점을 변경하지 않습니다. 조기·지연 적용, 위상 오류, X/Z 값은 !==로 검사합니다. 초기 리셋 이후 추가 리셋과 모든 중첩 요청·임의 주기는 시험하지 않습니다.

관찰은 상승 에지보다 1ns 늦습니다. 제18장과 달리 Driver는 pending 해제 후 다음 거래를 진행합니다. 시간 없는 UVM 거래 전달을 가정한 예상 시각입니다. 리셋 표본 6/16/26ns 3개, 정상 표본 36~336ns 31개(비활성 7개, 활성 24개)로 총 34개를 검사합니다.

CORRUPT_SAMPLE 부정 시험은 126ns(첫 듀티 2의 위상 2)에서 Monitor의 관찰 객체 pwm만 반전합니다. 정상 출력은 LOW이며 Scoreboard는 SB_PWM을 발생시켜야 합니다. DUT 자체를 변경하는 변이 시험과 구분합니다.

05전체 SystemVerilog/UVM 테스트벤치

tb/ch19/tb_pwm_monitor_agent_scoreboard.sv · SystemVerilog / UVM
`timescale 1ns/1ps
// 제19장: Sequence → Sequencer → Driver와 분리된 Monitor/Agent/Scoreboard.
// 원본 pwm_core.v의 포트·RTL은 수정하지 않는다.
// 입력 요청 2→6→0의 기대값은 Scoreboard가 독립적으로 관리한다.

interface pwm19_if(input logic clk);
    logic resetn = 1'b0;
    logic cfg_apply = 1'b0;
    logic [31:0] cfg_period = 32'd8;
    logic [31:0] cfg_duty = 32'd0;
    logic cfg_enable = 1'b0;
    logic cfg_polarity = 1'b0;
    wire pwm_out, pending, active_enable, active_polarity;
    wire [31:0] active_period, active_duty;
endinterface

package ch19_uvm_pkg;
    import uvm_pkg::*;
    `include "uvm_macros.svh"

    // (추가) Sequence가 Driver로 보내는 설정 거래.
    class pwm19_item extends uvm_sequence_item;
        bit [31:0] duty;
        `uvm_object_utils(pwm19_item)
        function new(string name="pwm19_item");
            super.new(name);
        endfunction
    endclass

    // (추가) Monitor가 관찰한 공개 신호를 담는 별도 거래.
    class pwm19_sample extends uvm_sequence_item;
        // (변경) 매 에지의 입력과 출력 모두 보존. X/Z도 유지한다.
        logic resetn, cfg_apply;
        logic [31:0] cfg_duty, cfg_period;
        logic cfg_enable, cfg_polarity;
        logic [31:0] duty, period;
        logic enable, polarity, pending, pwm;
        time sampled_at;
        `uvm_object_utils(pwm19_sample)
        function new(string name="pwm19_sample");
            super.new(name);
        endfunction
    endclass

    // (유지/확장) 제18장의 2, 6, 0 듀티 순서를 그대로 재사용.
    class pwm19_sequence extends uvm_sequence #(pwm19_item);
        `uvm_object_utils(pwm19_sequence)
        function new(string name="pwm19_sequence");
            super.new(name);
        endfunction
        task body();
            pwm19_item req;
            for (int i=0; i<3; i++) begin
                req = pwm19_item::type_id::create($sformatf("req_%0d", i));
                start_item(req);
                case (i)
                    0: req.duty = 32'd2;
                    1: req.duty = 32'd6;
                    default: req.duty = 32'd0;
                endcase
                finish_item(req);
            end
        endtask
    endclass

    class pwm19_sequencer extends uvm_sequencer #(pwm19_item);
        `uvm_component_utils(pwm19_sequencer)
        function new(string name, uvm_component parent);
            super.new(name, parent);
        endfunction
    endclass

    // (변경) Driver는 입력 신호만 구동하고 응답의 정답 판정은 하지 않는다.
    class pwm19_driver extends uvm_driver #(pwm19_item);
        `uvm_component_utils(pwm19_driver)
        virtual pwm19_if vif;
        int unsigned driven = 0;
        function new(string name, uvm_component parent);
            super.new(name, parent);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual pwm19_if)::get(this, "", "vif", vif))
                `uvm_fatal("NOVIF", "pwm19_if missing")
        endfunction
        task drive_one(pwm19_item req);
            int guard;
            @(negedge vif.clk);
            vif.cfg_period = 32'd8;
            vif.cfg_duty = req.duty;
            vif.cfg_enable = 1'b1;
            vif.cfg_polarity = 1'b0;
            vif.cfg_apply = 1'b1;
            @(negedge vif.clk);
            vif.cfg_apply = 1'b0;

            // 대기 신호를 통해 다음 요청을 언제 보낼지만 제어한다.
            // 실제 적용 듀티·출력 판정은 Monitor/Scoreboard의 책임.
            guard = 0;
            while (vif.pending !== 1'b0) begin
                @(negedge vif.clk);
                guard++;
                if (guard > 12)
                    `uvm_fatal("DRV_TIMEOUT", "pending was not cleared")
            end
            driven++;
        endtask
        task run_phase(uvm_phase phase);
            pwm19_item req;
            forever begin
                seq_item_port.get_next_item(req);
                drive_one(req);
                seq_item_port.item_done();
            end
        endtask
    endclass

    // (추가) Monitor는 DUT 출력/입력을 관찰할 뿐 신호를 구동하지 않는다.
    class pwm19_monitor extends uvm_monitor;
        `uvm_component_utils(pwm19_monitor)
        virtual pwm19_if vif;
        uvm_analysis_port #(pwm19_sample) ap;

        function new(string name, uvm_component parent);
            super.new(name, parent);
            ap = new("ap", this);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual pwm19_if)::get(this, "", "vif", vif))
                `uvm_fatal("NOVIF", "pwm19_if missing")
        endfunction
        task run_phase(uvm_phase phase);
            pwm19_sample sample;
            forever begin
                @(posedge vif.clk);
                #1; // DUT의 비차단 대입 완료 이후 공개 출력 관찰
                // (변경) 이벤트를 추론하지 않고 매 에지마다 새 객체를 전송한다.
                sample = pwm19_sample::type_id::create("cycle_sample");
                sample.resetn = vif.resetn;
                sample.cfg_apply = vif.cfg_apply;
                sample.cfg_duty = vif.cfg_duty;
                sample.cfg_period = vif.cfg_period;
                sample.cfg_enable = vif.cfg_enable;
                sample.cfg_polarity = vif.cfg_polarity;
                sample.duty = vif.active_duty;
                sample.period = vif.active_period;
                sample.enable = vif.active_enable;
                sample.polarity = vif.active_polarity;
                sample.pending = vif.pending;
                sample.pwm = vif.pwm_out;
                sample.sampled_at = $time;

    // (추가) Analysis Imp를 통해 Monitor의 관찰 거래를 받는다.
    // '정답 2→6→0'은 DUT에서 읽지 않고 검증 기준으로 별도 보관.
    class pwm19_scoreboard extends uvm_scoreboard;
        `uvm_component_utils(pwm19_scoreboard)
        uvm_analysis_imp #(pwm19_sample, pwm19_scoreboard) analysis_export;
        int unsigned requests = 0;
        int unsigned applied = 0;
        bit [31:0] planned_duty[3] = '{32'd2, 32'd6, 32'd0};
        time request_time[3];
        int unsigned reset_samples = 0, cycles = 0, samples = 0;
        int unsigned phase_idx = 0;
        bit expected_pending = 0, expected_enable = 0;
        bit [31:0] expected_duty = 0;
        bit expected_pwm;

        function new(string name, uvm_component parent);
            super.new(name, parent);
            analysis_export = new("analysis_export", this);
        endfunction

        // (변경) 예상 위상은 DUT의 pending/counter가 아니라 리셋 후
        // 관찰 클록 수의 나머지로 계산한다. 이 실습은 period=8 고정.
        function void write(pwm19_sample sample);
            if (sample.resetn === 1'b0) begin
                if (cycles != 0)
                    `uvm_fatal("SB_RESET", "Unexpected reset during transactions")
                reset_samples++;
                if (sample.period !== 32'd8 || sample.duty !== 32'd0 ||
                    sample.enable !== 1'b0 || sample.polarity !== 1'b0 ||
                    sample.pending !== 1'b0 || sample.pwm !== 1'b0)
                    `uvm_fatal("SB_RESET", "Reset output mismatch")
                return;
            end
            if (sample.resetn !== 1'b1 || reset_samples != 3)
                `uvm_fatal("SB_RESET", "Reset sequence mismatch")
            cycles++;
            phase_idx = cycles % 8;

            // 주기 경계에서는 이전 요청만 적용된다.
            if (phase_idx == 0 && expected_pending) begin
                if (applied >= requests || applied >= 3)
                    `uvm_fatal("SB_ORDER", "Apply without request")
                expected_duty = planned_duty[applied];
                expected_enable = 1;
                expected_pending = 0;
                if (sample.sampled_at - request_time[applied] > 80)
                    `uvm_fatal("SB_LATENCY", "Apply exceeded one period")
                applied++;
                `uvm_info("SB", $sformatf("APPLIED %0d duty=%0d",
                    applied, expected_duty), UVM_LOW)
            end
            // 요청의 예상 순서는 Driver의 거래 객체에 의존하지 않는다.
            if (sample.cfg_apply === 1'b1) begin
                if (requests >= 3 || expected_pending)
                    `uvm_fatal("SB_EXTRA", "Unexpected or overlapping request")
                if (sample.cfg_period !== 32'd8 ||
                    sample.cfg_duty !== planned_duty[requests] ||
                    sample.cfg_enable !== 1'b1 || sample.cfg_polarity !== 1'b0)
                    `uvm_fatal("SB_REQUEST", "Unexpected PWM request content")
                request_time[requests] = sample.sampled_at;
                requests++;
                expected_pending = 1;
                `uvm_info("SB", $sformatf("REQUEST %0d duty=%0d",
                    requests, sample.cfg_duty), UVM_LOW)
            end else if (sample.cfg_apply !== 1'b0)
                `uvm_fatal("SB_REQUEST", "Unknown cfg_apply")

            // 실제 공개 출력과 독립 기대값을 매 에지에서 대조한다.
            if (sample.period !== 32'd8 || sample.duty !== expected_duty ||
                sample.enable !== expected_enable || sample.polarity !== 1'b0 ||
                sample.pending !== expected_pending)
                `uvm_fatal("SB_STATE", $sformatf("t=%0d phase=%0d",
                    sample.sampled_at, phase_idx))
            expected_pwm = expected_enable && (phase_idx < expected_duty);
            if (sample.pwm !== expected_pwm)
                `uvm_fatal("SB_PWM", $sformatf("t=%0d phase=%0d expected=%0b got=%0b",
                    sample.sampled_at, phase_idx, expected_pwm, sample.pwm))
            if (expected_enable) samples++;
        endfunction
    endclass

            phase.raise_objection(this);
            repeat (3) @(negedge vif.clk);
            if (vif.active_period !== 32'd8 ||
                vif.active_duty !== 32'd0 || vif.pending !== 1'b0)
                `uvm_fatal("RESET", "PWM reset did not complete")
            vif.resetn = 1'b1;
            seq = pwm19_sequence::type_id::create("seq");
            seq.start(env.ag.sqr);
            seq_finished = 1;
            // (변경) 마지막 거래 적용 이후 전체 주기의 관찰까지 기다린다.
            wait (env.sb.samples == 24);
            @(negedge vif.clk);
            phase.drop_objection(this);
        endtask
        function void check_phase(uvm_phase phase);
            super.check_phase(phase);
            if (!seq_finished || env.ag.drv.driven != 3 ||
                env.sb.requests != 3 || env.sb.applied != 3 ||
                env.sb.samples != 24 || env.sb.cycles != 31 ||
                env.sb.reset_samples != 3 || env.sb.expected_pending)
                `uvm_fatal("SB_COUNT", "Missing or extra observations")
        endfunction
        function void report_phase(uvm_phase phase);
            super.report_phase(phase);
            if (uvm_report_server::get_server().get_severity_count(UVM_ERROR) != 0 ||
                uvm_report_server::get_server().get_severity_count(UVM_FATAL) != 0)
                `uvm_fatal("REPORT", "UVM error or fatal reported")
            `uvm_info("RESULT", "PASS CH19 requests=3 applied=3 samples=24", UVM_NONE)
        endfunction
    endclass
endpackage

06Ubuntu · Vivado XSIM 실행

제17~18장과 같은 Vivado 내장 UVM 라이브러리를 사용합니다. xvlog와 xelab에 -L uvm을 적용합니다. 설치 버전에서 UVM과 매크로 지원을 확인합니다.

scripts/run_ch19.sh · Bash
#!/usr/bin/env bash
# 제19장: Vivado XSIM 내장 UVM 라이브러리 사용.
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
RTL="${DUT_PATH:-$ROOT/reference_rtl/pwm_core.v}"
if [[ ! -f "$RTL" ]]; then
  echo "NOT_RUN: DUT missing" >&2
  exit 2
fi
RTL="$(realpath "$RTL")"
for tool in xvlog xelab xsim; do
  if ! command -v "$tool" >/dev/null 2>&1; then
    echo "NOT_RUN: $tool missing; source Vivado settings64.sh" >&2
    exit 2
  fi
done
mkdir -p "$ROOT/.build/ch19"
OUT="$(mktemp -d "$ROOT/.build/ch19/run.XXXXXX")"
cd "$OUT"
xvlog "$RTL" 2>&1 | tee rtl.log
xvlog -sv -L uvm "$ROOT/tb/ch19/tb_pwm_monitor_agent_scoreboard.sv" 2>&1 | tee tb.log
xelab -L uvm -debug typical tb_pwm_monitor_agent_scoreboard -s ch19_sim 2>&1 | tee elaborate.log
xsim ch19_sim -runall -onerror quit 2>&1 | tee normal.log
grep -Eq '\[RESULT\][[:space:]]+PASS CH19 requests=3 applied=3 samples=24[[:space:]]*$' normal.log
grep -Eq 'UVM_ERROR[[:space:]]*:[[:space:]]*0[[:space:]]*$' normal.log
grep -Eq 'UVM_FATAL[[:space:]]*:[[:space:]]*0[[:space:]]*$' normal.log
# 부정 시험: 126ns의 관찰 객체 PWM을 반전하여 Scoreboard를 시험한다.
# UVM $finish에 따른 종료 코드 차이가 있으므로 전용 진단도 검사한다.
rc=0
xsim ch19_sim -runall -onerror quit -testplusarg CORRUPT_SAMPLE > negative.log 2>&1 || rc=$?
grep -Eq 'UVM_FATAL.*\[SB_PWM\].*t=126 phase=2 expected=0 got=1' negative.log
if grep -q 'PASS CH19\|FAIL CH19 GLOBAL_TIMEOUT' negative.log; then
  echo "FAIL: unexpected negative result" >&2
  exit 1
fi
echo "PASS CH19 HARNESS (negative exit=$rc)"
echo "Logs: $OUT"
