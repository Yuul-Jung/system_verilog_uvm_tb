00개요 · 다중 Agent와 하드웨어 이벤트

제21장은 RAL 접근과 독립 Monitor/Predictor를 연결했습니다. 제22장은 버스 Agent(에이전트, 재사용 검증 구성 단위)와 Reset/IRQ Agent를 나누고 Virtual Sequence(버추얼 시퀀스, 다중 거래 제어)를 사용합니다. VIP(Verification IP, 검증 지식재산)는 여기서 직접 작성한 교육용 VIP-style 구성으로, AMD 상용 AXI VIP가 아닙니다.

설계 개념 연계: 제15·21장과 동일한 범용 AXI/RMAP 스냅샷입니다. amd_soc_p2 03.2장 IRQ·AXI, 04.2장 03~04절 및 10.3장 레지스터 사양과 개념 연계하며 현행 PWM/IRQ IP의 동일 RTL로 확인된 것은 아닙니다.

reference_rtl/axi4lite_slave_adapter.v · Verilog
module axi4lite_slave_adapter #(
    parameter integer C_S_AXI_ADDR_WIDTH = 32
)(
    input  wire                          s_axi_aclk,
    input  wire                          s_axi_aresetn,

    // AXI4-Lite Write Address Channel
    input  wire [C_S_AXI_ADDR_WIDTH-1:0] s_axi_awaddr,
    input  wire [2:0]                    s_axi_awprot,
    input  wire                          s_axi_awvalid,
    output wire                          s_axi_awready,

    // AXI4-Lite Write Data Channel
    input  wire [31:0]                   s_axi_wdata,
    input  wire [3:0]                    s_axi_wstrb,
    input  wire                          s_axi_wvalid,
    output wire                          s_axi_wready,

    // AXI4-Lite Write Response Channel
    output reg  [1:0]                    s_axi_bresp,
    output reg                           s_axi_bvalid,
    input  wire                          s_axi_bready,

    // AXI4-Lite Read Address Channel
    input  wire [C_S_AXI_ADDR_WIDTH-1:0] s_axi_araddr,
    input  wire [2:0]                    s_axi_arprot,
    input  wire                          s_axi_arvalid,
    output wire                          s_axi_arready,

    // AXI4-Lite Read Data Channel
    output reg  [31:0]                   s_axi_rdata,
    output reg  [1:0]                    s_axi_rresp,
    output reg                           s_axi_rvalid,
    input  wire                          s_axi_rready,

    // Bus-independent register access interface
    output wire                          wr_en,
    output wire [7:0]                    wr_offset,
    output wire [31:0]                   wr_data,
    output wire [3:0]                    wr_strb,

    output wire [7:0]                    rd_offset,
    input  wire [31:0]                   rd_data
);

// AXI response code used by this basic slave.
localparam [1:0] AXI_RESP_OKAY = 2'b00;

// Write Address and Write Data are independent AXI channels.
// Each channel is buffered separately until both are available.
reg                          aw_stored;
reg [C_S_AXI_ADDR_WIDTH-1:0] awaddr_stored;
reg                          w_stored;
reg [31:0]                   wdata_stored;
reg [3:0]                    wstrb_stored;

wire write_commit;

// Single-outstanding-write policy:
// Do not accept a new write while a response is pending.
assign s_axi_awready =
    s_axi_aresetn && !aw_stored && !s_axi_bvalid;

assign s_axi_wready =
    s_axi_aresetn && !w_stored && !s_axi_bvalid;

// Register-bank write occurs only after BOTH address and data
// channels have completed their independent handshakes.
assign write_commit =
    aw_stored && w_stored && !s_axi_bvalid;

assign wr_en     = write_commit;
assign wr_offset = awaddr_stored[7:0];
assign wr_data   = wdata_stored;
assign wr_strb   = wstrb_stored;

// Read side accepts one address only when no previous R response
// is waiting for the master.
assign s_axi_arready =
    s_axi_aresetn && !s_axi_rvalid;

// custom_register_bank read path is combinational.
// The address presented during AR handshake selects rd_data.
assign rd_offset = s_axi_araddr[7:0];

// AWPROT / ARPROT are not used by this educational register slave.
wire unused_prot;
assign unused_prot = ^{s_axi_awprot, s_axi_arprot};

always @(posedge s_axi_aclk) begin
    if (!s_axi_aresetn) begin
        aw_stored     <= 1'b0;
        awaddr_stored <= {C_S_AXI_ADDR_WIDTH{1'b0}};
        w_stored      <= 1'b0;
        wdata_stored  <= 32'h0000_0000;
        wstrb_stored  <= 4'b0000;
        s_axi_bresp   <= AXI_RESP_OKAY;
        s_axi_bvalid  <= 1'b0;
    end else begin
        // Capture Write Address independently.
        if (s_axi_awready && s_axi_awvalid) begin
            aw_stored     <= 1'b1;
            awaddr_stored <= s_axi_awaddr;
        end

        // Capture Write Data independently.
        if (s_axi_wready && s_axi_wvalid) begin
            w_stored     <= 1'b1;
            wdata_stored <= s_axi_wdata;
            wstrb_stored <= s_axi_wstrb;
        end

        // Once both parts exist, pulse wr_en through write_commit
        // and create the AXI B response.
        if (write_commit) begin
            aw_stored    <= 1'b0;
            w_stored     <= 1'b0;
            s_axi_bresp  <= AXI_RESP_OKAY;
            s_axi_bvalid <= 1'b1;
        end else if (s_axi_bvalid && s_axi_bready) begin
            s_axi_bvalid <= 1'b0;
        end
    end
end

always @(posedge s_axi_aclk) begin
    if (!s_axi_aresetn) begin
        s_axi_rdata  <= 32'h0000_0000;
        s_axi_rresp  <= AXI_RESP_OKAY;
        s_axi_rvalid <= 1'b0;
    end else begin
        // Capture the selected register value at AR handshake.
        if (s_axi_arready && s_axi_arvalid) begin
            s_axi_rdata  <= rd_data;
            s_axi_rresp  <= AXI_RESP_OKAY;
            s_axi_rvalid <= 1'b1;
        end else if (s_axi_rvalid && s_axi_rready) begin
            s_axi_rvalid <= 1'b0;
        end
    end
end

endmodule
reference_rtl/custom_register_bank.v · Verilog
module custom_register_bank (
    input  wire        clk,
    input  wire        resetn,

    // Bus-independent register access interface.
    // SoC 12 will connect this interface to AXI4-Lite.
    input  wire        wr_en,
    input  wire [7:0]  wr_offset,
    input  wire [31:0] wr_data,
    input  wire [3:0]  wr_strb,

    input  wire [7:0]  rd_offset,
    output reg  [31:0] rd_data,

    // Hardware status / event inputs
    input  wire        busy_i,
    input  wire        done_i,
    input  wire        error_i,
    input  wire        irq_done_event_i,
    input  wire        irq_error_event_i,
    input  wire [31:0] data_out_i,

    // Control outputs
    output reg         enable_o,
    output reg  [1:0]  mode_o,
    output reg         start_pulse_o,
    output reg         clear_pulse_o,
    output reg  [31:0] data_in_o,
    output reg  [1:0]  irq_status_o
);

localparam [7:0] REG_CONTROL    = 8'h00;
localparam [7:0] REG_STATUS     = 8'h04;
localparam [7:0] REG_COMMAND    = 8'h08;
localparam [7:0] REG_IRQ_STATUS = 8'h0C;
localparam [7:0] REG_DATA_IN    = 8'h10;
localparam [7:0] REG_DATA_OUT   = 8'h14;
localparam [7:0] REG_VERSION    = 8'h18;
localparam [7:0] REG_ID         = 8'h1C;

localparam [31:0] VERSION_VALUE = 32'h0001_0000;
localparam [31:0] ID_VALUE      = 32'h524D_4150; // ASCII "RMAP"

integer byte_index;

always @(posedge clk) begin
    if (!resetn) begin
        enable_o      <= 1'b0;
        mode_o        <= 2'b00;
        start_pulse_o <= 1'b0;
        clear_pulse_o <= 1'b0;
        data_in_o     <= 32'h0000_0000;
        irq_status_o  <= 2'b00;
    end else begin
        // W1P pulses are one clock wide by default.
        start_pulse_o <= 1'b0;
        clear_pulse_o <= 1'b0;

        // Hardware events set IRQ status.
        // Event set has priority over software clear in the same cycle.
        if (irq_done_event_i) begin
            irq_status_o[0] <= 1'b1;
        end else if (
            wr_en &&
            (wr_offset == REG_IRQ_STATUS) &&
            wr_strb[0] &&
            wr_data[0]
        ) begin
            irq_status_o[0] <= 1'b0;
        end

        if (irq_error_event_i) begin
            irq_status_o[1] <= 1'b1;
        end else if (
            wr_en &&
            (wr_offset == REG_IRQ_STATUS) &&
            wr_strb[0] &&
            wr_data[1]
        ) begin
            irq_status_o[1] <= 1'b0;
        end

        if (wr_en) begin
            case (wr_offset)
                REG_CONTROL: begin
                    if (wr_strb[0]) begin
                        enable_o <= wr_data[0];
                        mode_o   <= wr_data[2:1];
                    end
                end

                REG_COMMAND: begin
                    if (wr_strb[0]) begin
                        start_pulse_o <= wr_data[0];
                        clear_pulse_o <= wr_data[1];
                    end
                end

                REG_DATA_IN: begin
                    // Byte-enable write using WSTRB-equivalent strobes.
                    for (byte_index = 0; byte_index < 4; byte_index = byte_index + 1) begin
                        if (wr_strb[byte_index]) begin
                            data_in_o[byte_index*8 +: 8] <=
                                wr_data[byte_index*8 +: 8];
                        end
                    end
                end

                default: begin
                    // RO / reserved / unmapped writes have no effect.
                end
            endcase
        end
    end
end

always @(*) begin
    case (rd_offset)
        REG_CONTROL: begin
            rd_data = {29'h0, mode_o, enable_o};
        end

        REG_STATUS: begin
            rd_data = {29'h0, error_i, done_i, busy_i};
        end

        REG_COMMAND: begin
            // W1P command register has no retained state.
            rd_data = 32'h0000_0000;
        end

        REG_IRQ_STATUS: begin
            rd_data = {30'h0, irq_status_o};
        end

        REG_DATA_IN: begin
            rd_data = data_in_o;
        end

        REG_DATA_OUT: begin
            rd_data = data_out_i;
        end

        REG_VERSION: begin
            rd_data = VERSION_VALUE;
        end

        REG_ID: begin
            rd_data = ID_VALUE;
        end

        default: begin
            rd_data = 32'h0000_0000;
        end
    endcase
end

endmodule

AW/W 수락 다음 상승 에지에 bank가 wr_en을 읽어 쓰기를 반영합니다. IRQ_STATUS의 bit 0은 done, bit 1은 error이며 각 비트에서 하드웨어 이벤트 set이 소프트웨어 W1C(Write One to Clear, 1 쓰기 해제)보다 우선합니다. IRQ 읽기는 AR 수락 시점의 조합 읽기값을 캡처합니다. B/R 응답 시점과 실제 레지스터 상태 변경 시점을 혼동하지 않습니다.

AXI Driver는 하강 에지에서 입력을 구동하고 독립 Monitor는 상승 에지에서 실제 핸드셰이크를 관찰합니다. B/R 응답을 두 클록 대기시키며 각 상승 에지에서 수락 시점까지 응답과 데이터가 유지되는지 비교합니다. X/Z는 2상태 거래 객체에 복사하기 전에 검사합니다. 현재 시험은 단일 미결 쓰기·읽기를 순차 수행하고, 별도 IRQ 이벤트 Agent와의 동시성을 시험합니다. AXI 자체의 모든 병렬 채널 조합이나 상용 VIP의 기능 전체를 검증하는 것이 아닙니다.

Event Monitor는 상승 에지에서 실제 bank 입력과 이벤트를 저장하고 #1ns 뒤 irq_status를 읽습니다. Scoreboard는 매 클록 기존 기대 IRQ에서 clear_mask를 제거하고 set_mask를 OR 합니다.

기대값 계산 · SystemVerilog
expected_irq=(expected_irq & ~clear_mask) | set_mask;

wr_en은 실제 bank 공개 입력입니다. 관찰된 commit의 offset/data/strb를 큐에 보관하고 AXI의 B 완료 거래와 대조하여 누락·추가·payload 차이를 검출합니다. IRQ의 기대값은 DUT의 irq_status에서 얻지 않습니다. 다만 commit 시점 자체는 관찰된 bank 입력을 기준으로 하므로 모든 commit 타이밍 결함을 독립적으로 예측하는 모델은 아닙니다. CONTROL/DATA_IN 기대값은 순차 B 완료 모델을 유지합니다.

IRQ 상태는 AR 수락 에지 직전 기대값을 irq_before[time]에 저장하고 R 수락 시 ar_stamp로 조회합니다. 따라서 R 응답 대기 시점의 현재 IRQ를 잘못 비교하지 않습니다. 이번 입력에서는 AR/R 사이 별도 이벤트 경합까지 직접 자극하지는 않습니다.

04검증 시나리오 · 6개 쓰기와 13개 읽기

충돌 시험은 Event Driver가 실제 wr_en 및 IRQ 주소와 데이터가 유효한 하강 에지에 done/error를 함께 올리고 다음 하강 에지에서 내립니다. 그 사이 한 상승 에지에서 clear와 set이 동시에 발생합니다. Scoreboard가 collisions=1로 실제 충돌을 확인합니다. IRQ 이벤트 비트 발생 총 4개(DONE 1, ERROR 1, 동시 2), 리셋 구간 2개(초기 및 재리셋), commit 6개, zero_strobes=1입니다.

부정 시험 CORRUPT_IRQ는 충돌 시 관찰 객체의 bit 0만 반전합니다. 예상 11, 관찰 10으로 IRQ_MISMATCH 진단을 발생시킵니다. 원본 RTL 변이 시험이 아니라 비교 경로의 오류 검출 시험입니다.

`timescale 1ns/1ps
// 제22장: AXI4-Lite Master Agent + Reset/IRQ Agent + Virtual Sequence.
// 상용 AMD AXI VIP가 아닌 교육용 VIP-style(재사용 검증 컴포넌트) 구현.
// 기존 axi4lite_slave_adapter.v 및 custom_register_bank.v는 수정하지 않는다.

// (추가) 버스 전용 Interface: AXI 신호만 묶는다.
interface axi22_if(input logic clk, input logic resetn);
    logic [31:0] awaddr=0, wdata=0, araddr=0;
    logic [3:0] wstrb=0;
    logic awvalid=0, wvalid=0, bready=0, arvalid=0, rready=0;
    wire awready, wready, bvalid, arready, rvalid;
    wire [1:0] bresp, rresp;
    wire [31:0] rdata;
endinterface

// (추가) 하드웨어 이벤트 전용 Interface: AXI와 구동 책임을 분리한다.
interface irq22_if(input logic clk);
    logic resetn=0;
    logic done_event=0, error_event=0;
    wire [1:0] irq_status;
    // (추가) 공개 레지스터 뱅크 입력: 실제 commit 경계 관찰 및 충돌 자극.
    wire wr_en;
    wire [7:0] wr_offset;
    wire [31:0] wr_data;
    wire [3:0] wr_strb;
endinterface

package ch22_pkg;
    import uvm_pkg::*;
    `include "uvm_macros.svh"

    // 두 개의 독립 Analysis Imp(write_bus/write_irq)를 선언한다.
    `uvm_analysis_imp_decl(_bus)
    `uvm_analysis_imp_decl(_irq)

    typedef enum {EV_RELEASE, EV_ASSERT_RESET, EV_DONE, EV_ERROR, EV_BOTH_ON_CLEAR} evt22_kind_e;

    class irq22_sample extends uvm_sequence_item;

    class axi22_sequencer extends uvm_sequencer #(axi22_item);
        `uvm_component_utils(axi22_sequencer)
        function new(string name, uvm_component parent);
            super.new(name,parent);
        endfunction
    endclass

    class irq22_sequencer extends uvm_sequencer #(irq22_item);
        `uvm_component_utils(irq22_sequencer)
        function new(string name, uvm_component parent);
            super.new(name,parent);
        endfunction
    endclass

    // AXI Master Driver: AW/W 독립 Handshake와 B/R Backpressure를 보존한다.
    class axi22_driver extends uvm_driver #(axi22_item);
        `uvm_component_utils(axi22_driver)
        virtual axi22_if vif;
        int unsigned driven=0;
        function new(string name,uvm_component parent);
            super.new(name,parent);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual axi22_if)::get(this,"","vif",vif))
                `uvm_fatal("AXI_VIF","AXI interface missing")
        endfunction
        task automatic write_one(axi22_item t);
            bit aw_done=0, w_done=0;
            @(negedge vif.clk);
            vif.awaddr=t.addr; vif.awvalid=1'b1;
            vif.wdata=t.data; vif.wstrb=t.strb; vif.wvalid=1'b1;
            for (int k=0;k<100;k++) begin
                @(posedge vif.clk);
                if (vif.awvalid && vif.awready) aw_done=1;
                if (vif.wvalid && vif.wready) w_done=1;
                @(negedge vif.clk);
                if (aw_done) vif.awvalid=1'b0;
                if (w_done) vif.wvalid=1'b0;
                if (aw_done && w_done) break;
            end
            if (!aw_done || !w_done)
                `uvm_fatal("AXI_AW_W","Write handshake timeout")
            for (int k=0;k<100;k++) begin
                @(negedge vif.clk);
                if (vif.bvalid) begin
                    t.resp=vif.bresp;
                    // 두 클록 동안 BREADY=0 → 응답 유지 후 수락.
                    repeat (2) begin
                        @(negedge vif.clk);
                        if (vif.bvalid !== 1'b1 || vif.bresp !== t.resp)
                            `uvm_fatal("B_HOLD","Write response changed during stall")
                    end
                    vif.bready=1'b1;
                    @(posedge vif.clk);
                    @(negedge vif.clk);
                    vif.bready=1'b0;
                    return;
                end
            end
            `uvm_fatal("B_TIMEOUT","Write response timeout")
        endtask
        task automatic read_one(axi22_item t);
            bit ar_done=0;
            @(negedge vif.clk);
            vif.araddr=t.addr; vif.arvalid=1'b1;
            for (int k=0;k<100;k++) begin
                @(posedge vif.clk);
                if (vif.arvalid && vif.arready) ar_done=1;
                @(negedge vif.clk);
                if (ar_done) begin vif.arvalid=1'b0; break; end
            end
            if (!ar_done) `uvm_fatal("AR_TIMEOUT","Read address timeout")
            for (int k=0;k<100;k++) begin
                @(negedge vif.clk);
                if (vif.rvalid) begin
                    t.data=vif.rdata; t.resp=vif.rresp;
                    // 두 클록 동안 RREADY=0 → 데이터 유지 검사.
                    repeat (2) begin
                        @(negedge vif.clk);
                        if (vif.rvalid !== 1'b1 || vif.rdata !== t.data ||
                            vif.rresp !== t.resp)
                            `uvm_fatal("R_HOLD","Read response changed during stall")
                    end
                    vif.rready=1'b1;
                    @(posedge vif.clk);
                    @(negedge vif.clk);
                    vif.rready=1'b0;
                    return;
                end
            end
            `uvm_fatal("R_TIMEOUT","Read response timeout")
        endtask
        task run_phase(uvm_phase phase);
            axi22_item req;
            forever begin
                seq_item_port.get_next_item(req);
                if (req.is_write) write_one(req);
                else read_one(req);
                driven++;
                seq_item_port.item_done();
            end
        endtask
    endclass

    // AXI Monitor는 Driver 완료 이벤트를 쓰지 않고 실제 5채널을 관찰한다.
    class axi22_monitor extends uvm_monitor;
        `uvm_component_utils(axi22_monitor)
        virtual axi22_if vif;
        uvm_analysis_port #(axi22_item) ap;
        bit aw_seen=0, w_seen=0, ar_seen=0;
        bit [31:0] saved_aw, saved_wd, saved_ar;
        bit [3:0] saved_strb;
        time saved_ar_stamp;
        bit b_held=0, r_held=0;
        logic [1:0] held_bresp, held_rresp;
        logic [31:0] held_rdata;
        function new(string name,uvm_component parent);
            super.new(name,parent);
            ap=new("ap",this);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual axi22_if)::get(this,"","vif",vif))
                `uvm_fatal("AXI_VIF","AXI monitor interface missing")
        endfunction
        task run_phase(uvm_phase phase);
            axi22_item s;
            forever begin
                @(posedge vif.clk); // DUT가 Handshake를 읽는 에지에서 관찰
                if (!vif.resetn) begin
                    aw_seen=0; w_seen=0; ar_seen=0;
                    b_held=0; r_held=0;
                end else begin
                    if ((^{vif.resetn,vif.awvalid,vif.awready,vif.wvalid,vif.wready,
                           vif.bvalid,vif.bready,vif.arvalid,vif.arready,
                           vif.rvalid,vif.rready}) === 1'bx)
                        `uvm_fatal("BUS_X","Unknown handshake")
                    // (추가) 모든 상승 에지에서 응답 대기 및 수락 에지까지 유지 검사.
                    if (b_held && (vif.bvalid !== 1'b1 || vif.bresp !== held_bresp))
                        `uvm_fatal("B_HOLD","B changed while pending")
                    if (r_held && (vif.rvalid !== 1'b1 || vif.rresp !== held_rresp ||
                                   vif.rdata !== held_rdata))
                        `uvm_fatal("R_HOLD","R changed while pending")
                    b_held=vif.bvalid && !vif.bready;
                    r_held=vif.rvalid && !vif.rready;
                    if (b_held) held_bresp=vif.bresp;
                    if (r_held) begin held_rresp=vif.rresp; held_rdata=vif.rdata; end
                    if (vif.awvalid && vif.awready) begin
                        if (aw_seen || (^vif.awaddr) === 1'bx)
                            `uvm_fatal("MON_AW","Duplicate or unknown AW")
                        saved_aw=vif.awaddr; aw_seen=1;
                    end
                    if (vif.wvalid && vif.wready) begin
                        if (w_seen || (^{vif.wdata,vif.wstrb}) === 1'bx)
                            `uvm_fatal("MON_W","Duplicate or unknown W")
                        saved_wd=vif.wdata; saved_strb=vif.wstrb; w_seen=1;
                    end
                    if (vif.arvalid && vif.arready) begin
                        if (ar_seen || (^vif.araddr) === 1'bx)
                            `uvm_fatal("MON_AR","Duplicate or unknown AR")
                        saved_ar=vif.araddr; saved_ar_stamp=$time; ar_seen=1;
                    end
                    if (vif.bvalid && vif.bready) begin
                        if (!aw_seen || !w_seen || vif.bresp !== 2'b00)
                            `uvm_fatal("AXI_MON_B","B before AW/W Handshake")
                        s=axi22_item::type_id::create("observed_write");
                        s.is_write=1; s.addr=saved_aw; s.data=saved_wd;
                        s.strb=saved_strb; s.resp=vif.bresp;
                        ap.write(s);
                        aw_seen=0; w_seen=0;
                    end
                    if (vif.rvalid && vif.rready) begin
                        if (!ar_seen || vif.rresp !== 2'b00 || (^vif.rdata) === 1'bx)
                            `uvm_fatal("AXI_MON_R","R before AR Handshake")
                        s=axi22_item::type_id::create("observed_read");
                        s.is_write=0; s.addr=saved_ar; s.data=vif.rdata;
                        s.ar_stamp=saved_ar_stamp;
                        s.resp=vif.rresp;
                        ap.write(s);
                        ar_seen=0;
                    end
                end
            end
        endtask
    endclass

    // 별도 Event Driver는 Reset/IRQ 입력만 구동한다.
    class irq22_driver extends uvm_driver #(irq22_item);
        `uvm_component_utils(irq22_driver)
        virtual irq22_if vif;
        int unsigned events_driven=0;
        function new(string name,uvm_component parent);
            super.new(name,parent);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual irq22_if)::get(this,"","vif",vif))
                `uvm_fatal("IRQ_VIF","IRQ interface missing")
        endfunction
        task run_phase(uvm_phase phase);
            irq22_item req;
            forever begin
                seq_item_port.get_next_item(req);
                case (req.kind)
                    EV_RELEASE: begin
                        @(negedge vif.clk);
                        vif.resetn=1'b1;
                    end
                    EV_ASSERT_RESET: begin
                        @(negedge vif.clk);
                        vif.resetn=1'b0;
                        repeat (3) @(negedge vif.clk);
                    end
                    EV_DONE: begin
                        @(negedge vif.clk);
                        vif.done_event=1'b1;
                        @(negedge vif.clk);
                        vif.done_event=1'b0;
                    end
                    EV_ERROR: begin
                        @(negedge vif.clk);
                        vif.error_event=1'b1;
                        @(negedge vif.clk);
                        vif.error_event=1'b0;
                    end
                    // (추가) 실제 bank 쓰기 경계에 두 이벤트를 정렬한다.
                    EV_BOTH_ON_CLEAR: begin
                        int guard=0;
                        do begin
                            @(negedge vif.clk);
                            guard++;
                            if (guard>100) `uvm_fatal("COLLISION_TIMEOUT","No W1C commit")
                        end while (!(vif.wr_en === 1'b1 && vif.wr_offset === 8'h0c &&
                                     vif.wr_strb[0] === 1'b1 && vif.wr_data[1:0] === 2'b11));
                        vif.done_event=1'b1; vif.error_event=1'b1;
                        @(negedge vif.clk);
                        vif.done_event=1'b0; vif.error_event=1'b0;
                    end
                endcase
                events_driven++;
                seq_item_port.item_done();
            end
        endtask
    endclass

    // 별도 Event Monitor: IRQ 상태·Reset을 관찰해 Scoreboard에 전달.
    class irq22_monitor extends uvm_monitor;
        `uvm_component_utils(irq22_monitor)
        virtual irq22_if vif;
        uvm_analysis_port #(irq22_sample) ap;
        function new(string name,uvm_component parent);
            super.new(name,parent);
            ap=new("ap",this);
        endfunction
        function void build_phase(uvm_phase phase);
            super.build_phase(phase);
            if (!uvm_config_db#(virtual irq22_if)::get(this,"","vif",vif))
                `uvm_fatal("IRQ_VIF","IRQ monitor interface missing")
        endfunction
        task run_phase(uvm_phase phase);
            irq22_sample s;
            forever begin
                @(posedge vif.clk);
                // (변경) 입력은 상승 에지, 출력은 NBA 이후에 같은 객체로 관찰.
                s=irq22_sample::type_id::create("cycle");
                s.stamp=$time; s.resetn=vif.resetn;
                s.done_event=vif.done_event; s.error_event=vif.error_event;
                s.wr_en=vif.wr_en; s.offset=vif.wr_offset;
                s.data=vif.wr_data; s.strb=vif.wr_strb;
                #1;
                s.status=vif.irq_status;
                if ($test$plusargs("CORRUPT_IRQ") && s.resetn &&
                    s.done_event && s.error_event && s.wr_en)
                    s.status=s.status ^ 2'b01;
                ap.write(s);
            end
        endtask
    endclass

    // 버스 완료 관찰과 IRQ 이벤트 관찰을 별도 Analysis Imp로 수집.
    class axi22_scoreboard extends uvm_scoreboard;
        `uvm_component_utils(axi22_scoreboard)
        uvm_analysis_imp_bus #(axi22_item,axi22_scoreboard) bus_in;
        uvm_analysis_imp_irq #(irq22_sample,axi22_scoreboard) irq_in;
        bit [31:0] expected_control=0, expected_data=0;
        bit [1:0] expected_irq=0;
        int unsigned writes=0, reads=0, events=0, resets=0;
        int unsigned collisions=0, zero_strobes=0, commits=0;
        bit previous_resetn=1;
        bit [1:0] irq_before[time];
        axi22_item committed[$];
        function new(string name,uvm_component parent);
            super.new(name,parent);
            bus_in=new("bus_in",this);
            irq_in=new("irq_in",this);
        endfunction

        function void write_bus(axi22_item s);
            bit [31:0] exp_data;
            axi22_item c;
            if (s.resp !== 2'b00)
                `uvm_fatal("AXI_RESP","Non-OKAY AXI response")
            if (s.is_write) begin
                if (committed.size()==0)
                    `uvm_fatal("COMMIT","B without register-bank commit")
                c=committed.pop_front();
                if (s.addr[7:0] !== c.addr[7:0] || s.data !== c.data ||
                    s.strb !== c.strb)
                    `uvm_fatal("COMMIT","AXI transaction differs from bank input")
                case (s.addr)
                    'h00: if (s.strb[0])
                        expected_control={29'd0,s.data[2:0]};
                    'h10: for (int i=0;i<4;i++)
                        if (s.strb[i])
                            expected_data[i*8 +: 8]=s.data[i*8 +: 8];
                    'h0c: ; // IRQ는 B 응답이 아니라 실제 commit 에지에서 갱신한다.
                    default: `uvm_fatal("AXI_WR_ADDR","Unexpected write address")
                endcase
                writes++;
            end else begin
                case (s.addr)
                    'h00: exp_data=expected_control;
                    'h0c: begin
                        if (!irq_before.exists(s.ar_stamp))
                            `uvm_fatal("AR_TIME","No IRQ state at AR acceptance")
                        exp_data={30'd0,irq_before[s.ar_stamp]};
                    end
                    'h10: exp_data=expected_data;
                    'h18: exp_data=32'h00010000;
                    'h1c: exp_data=32'h524D4150;
                    default: `uvm_fatal("AXI_RD_ADDR","Unexpected read address")
                endcase
                if (s.data !== exp_data)
                    `uvm_fatal("AXI_READ_MISMATCH",
                        $sformatf("addr=%h got=%h expected=%h",
                                  s.addr,s.data,exp_data))
                reads++;
            end
        endfunction

        function void write_irq(irq22_sample s);
            axi22_item c;
            bit [1:0] clear_mask, set_mask;
            irq_before[s.stamp]=expected_irq;
            if (s.resetn === 1'b0) begin
                if (previous_resetn) resets++;
                if (committed.size()!=0)
                    `uvm_fatal("RESET_PENDING","Unexpected pending bank commit")
                expected_irq=0; expected_control=0; expected_data=0;
                previous_resetn=0;
            end else begin
                if ((^{s.resetn,s.done_event,s.error_event,s.wr_en}) === 1'bx)
                    `uvm_fatal("IRQ_X","Unknown bank control")
                previous_resetn=1;
                clear_mask=0; set_mask={s.error_event,s.done_event};
                if (s.wr_en) begin
                    if ((^{s.offset,s.data,s.strb}) === 1'bx)
                        `uvm_fatal("COMMIT_X","Unknown bank write")
                    c=axi22_item::type_id::create("commit");
                    c.addr={24'd0,s.offset}; c.data=s.data; c.strb=s.strb;
                    committed.push_back(c); commits++;
                    if (s.offset==8'h0c) begin
                        if (s.strb[0]) clear_mask=s.data[1:0];
                        else zero_strobes++;
                    end
                end
                if (s.done_event) events++;
                if (s.error_event) events++;
                if (clear_mask==2'b11 && set_mask==2'b11) collisions++;
                // (추가) 동일 비트의 set이 clear보다 우선.
                expected_irq=(expected_irq & ~clear_mask) | set_mask;
            end
            if (s.status !== expected_irq)
                `uvm_fatal("IRQ_MISMATCH",$sformatf(
                    "status=%b expected=%b collisions=%0d",
                    s.status,expected_irq,collisions))
        endfunction
    endclass

    // Virtual Sequence 하나가 서로 다른 두 Agent를 순서대로 조정한다.
    class axi22_system_sequence extends uvm_sequence #(uvm_sequence_item);
        `uvm_object_utils(axi22_system_sequence)
        `uvm_declare_p_sequencer(axi22_virtual_sequencer)
        function new(string name="axi22_system_sequence");
            super.new(name);
        endfunction

        task automatic ev(evt22_kind_e kind);
            irq22_event_sequence s=irq22_event_sequence::type_id::create("ev");
            s.kind=kind;
            s.start(p_sequencer.evt_sqr);
        endtask

        task automatic wr(bit [31:0] a,bit [31:0] d,bit [3:0] strb=4'hf);
            axi22_bus_sequence s=axi22_bus_sequence::type_id::create("write");
            s.is_write=1; s.addr=a; s.send_data=d; s.strb=strb;
            s.start(p_sequencer.bus_sqr);
            if (s.return_resp !== 2'b00)
                `uvm_fatal("WRITE","Write response was not OKAY")
        endtask

        task automatic rd(bit [31:0] a,bit [31:0] expected);
            axi22_bus_sequence s=axi22_bus_sequence::type_id::create("read");
            s.is_write=0; s.addr=a;
            s.start(p_sequencer.bus_sqr);
            if (s.return_resp !== 2'b00 || s.return_data !== expected)
                `uvm_fatal("READ",
                    $sformatf("addr=%h got=%h expected=%h",
                              a,s.return_data,expected))
        endtask

        task body();
            // 초기 리셋 → 제어/데이터 레지스터 접근.
            ev(EV_RELEASE);
            wr('h00,32'h00000005);
            wr('h10,32'h11223344);
            rd('h00,32'h00000005);
            rd('h10,32'h11223344);

            // 실제 HW IRQ 이벤트 발생 → AXI Read로 관찰.
            ev(EV_DONE);
            rd('h0c,32'h00000001);
            ev(EV_ERROR);
            rd('h0c,32'h00000003);

            // W1C(Write 1 to Clear, 1을 써서 상태 비트 해제) 시험.
            wr('h0c,32'h00000001);
            rd('h0c,32'h00000002);
            wr('h0c,32'h00000002);
            rd('h0c,32'h00000000);

            // (추가) 두 Agent를 동시에 실행하여 같은 commit 에지에서 set/clear 충돌.
            fork
                wr('h0c,32'h00000003);
                ev(EV_BOTH_ON_CLEAR);
            join
            rd('h0c,32'h00000003);
            // WSTRB[0]=0: W1C가 무시되어 두 비트가 유지된다.
            wr('h0c,32'h00000003,4'b0000);
            rd('h0c,32'h00000003);

            // RO 레지스터와 리셋 후 상태 초기화 확인.
            rd('h18,32'h00010000);
            rd('h1c,32'h524D4150);
            ev(EV_ASSERT_RESET);
            ev(EV_RELEASE);
            rd('h0c,32'h00000000);
            rd('h00,32'h00000000);
            rd('h10,32'h00000000);
        endtask
    endclass

    class axi22_test extends uvm_test;

        task run_phase(uvm_phase phase);

                env.bus_agent.drv.driven!=19 ||
                env.event_agent.drv.events_driven!=6 ||
                env.bus_agent.mon.aw_seen || env.bus_agent.mon.w_seen ||
                env.bus_agent.mon.ar_seen)
                `uvm_fatal("COUNT","Incomplete multi-agent verification")
        endfunction
        function void report_phase(uvm_phase phase);
            super.report_phase(phase);
            if (uvm_report_server::get_server().get_severity_count(UVM_ERROR)!=0 ||
                uvm_report_server::get_server().get_severity_count(UVM_FATAL)!=0)
                `uvm_fatal("REPORT","UVM error or fatal reported")
            `uvm_info("RESULT","PASS CH22 AXI_W=6 AXI_R=13 IRQ=4 RESET=2 COLLISION=1 ZERO=1",UVM_NONE)
        endfunction
    endclass

#!/usr/bin/env bash
# 제22장: Vivado XSIM 내장 UVM 라이브러리 사용.
set -euo pipefail
ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
RTL="${RTL_DIR:-$ROOT/reference_rtl}"
for f in axi4lite_slave_adapter.v custom_register_bank.v; do
  if [[ ! -f "$RTL/$f" ]]; then echo "NOT_RUN: missing $RTL/$f" >&2; exit 2; fi
done
RTL="$(realpath "$RTL")"
for tool in xvlog xelab xsim; do
  if ! command -v "$tool" >/dev/null 2>&1; then
    echo "NOT_RUN: $tool missing; source Vivado settings64.sh" >&2
    exit 2
  fi
done
mkdir -p "$ROOT/.build/ch22"
OUT="$(mktemp -d "$ROOT/.build/ch22/run.XXXXXX")"
cd "$OUT"
xvlog "$RTL/axi4lite_slave_adapter.v" "$RTL/custom_register_bank.v" 2>&1 | tee rtl.log
xvlog -sv -L uvm "$ROOT/tb/ch22/tb_axi_multiagent_irq.sv" 2>&1 | tee tb.log
xelab -timescale 1ns/1ps -L uvm -debug typical tb_axi_multiagent_irq -s ch22_sim 2>&1 | tee elaborate.log
xsim ch22_sim -runall -onerror quit 2>&1 | tee normal.log
grep -Eq '\[RESULT\][[:space:]]+PASS CH22 AXI_W=6 AXI_R=13 IRQ=4 RESET=2 COLLISION=1 ZERO=1[[:space:]]*$' normal.log
grep -Eq 'UVM_ERROR[[:space:]]*:[[:space:]]*0[[:space:]]*$' normal.log
grep -Eq 'UVM_FATAL[[:space:]]*:[[:space:]]*0[[:space:]]*$' normal.log
rc=0
xsim ch22_sim -runall -onerror quit -testplusarg CORRUPT_IRQ > negative.log 2>&1 || rc=$?
grep -Eq 'UVM_FATAL.*\[IRQ_MISMATCH\].*status=10 expected=11 collisions=1' negative.log
if grep -Eq 'PASS CH22|CH22 GLOBAL_TIMEOUT' negative.log; then
  echo "FAIL: unexpected negative result" >&2
  exit 1
fi
echo "PASS CH22 HARNESS (negative exit=$rc)"
echo "Logs: $OUT"
